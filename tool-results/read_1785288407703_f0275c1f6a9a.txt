    78→// ─── GRAPH VIEW ───
    79→function GraphView() {
    80→  const svgRef = useRef<SVGSVGElement>(null);
    81→  const [hoveredNode, setHoveredNode] = useState<string | null>(null);
    82→  const [viewBox, setViewBox] = useState("-5 -5 110 90");
    83→  const [isDragging, setIsDragging] = useState(false);
    84→  const dragStart = useRef({ x: 0, y: 0 });
    85→  const vbStart = useRef({ x: -5, y: -5 });
    86→
    87→  const nations = useHegemoniaStore((s) => s.nations);
    88→  const edges = useHegemoniaStore((s) => s.edges);
    89→  const selectedNationId = useHegemoniaStore((s) => s.selectedNationId);
    90→  const setSelectedNation = useHegemoniaStore((s) => s.setSelectedNation);
    91→  const activeTab = useHegemoniaStore((s) => s.activeTab);
    92→  const actionTargetId = useHegemoniaStore((s) => s.actionTargetId);
    93→  const setActionTarget = useHegemoniaStore((s) => s.setActionTarget);
    94→
    95→  const maxVolume = useMemo(() => Math.max(1, ...edges.map((e) => e.volume)), [edges]);
    96→
    97→  const handleWheel = useCallback((e: React.WheelEvent) => {
    98→    e.preventDefault();
    99→    const [vx, vy, vw, vh] = viewBox.split(" ").map(Number);
   100→    const svg = svgRef.current;
   101→    if (!svg) return;
   102→    const rect = svg.getBoundingClientRect();
   103→    const mx = ((e.clientX - rect.left) / rect.width) * vw + vx;
   104→    const my = ((e.clientY - rect.top) / rect.height) * vh + vy;
   105→    const f = e.deltaY > 0 ? 1.1 : 0.9;
   106→    const nvw = Math.max(30, Math.min(130, vw * f));
   107→    const nvh = nvw * (vh / vw);
   108→    const r = nvw / vw;
   109→    setViewBox(`${(mx - (mx - vx) * r).toFixed(1)} ${(my - (my - vy) * r).toFixed(1)} ${nvw.toFixed(1)} ${nvh.toFixed(1)}`);
   110→  }, [viewBox]);
   111→
   112→  const handleMouseDown = useCallback((e: React.MouseEvent) => {
   113→    if ((e.target as Element).closest(".gn")) return;
   114→    setIsDragging(true);
   115→    dragStart.current = { x: e.clientX, y: e.clientY };
   116→    const [vx, vy] = viewBox.split(" ").map(Number);
   117→    vbStart.current = { x: vx, y: vy };
   118→  }, [viewBox]);
   119→
   120→  const handleMouseMove = useCallback((e: React.MouseEvent) => {
   121→    if (!isDragging) return;
   122→    const svg = svgRef.current;
   123→    if (!svg) return;
   124→    const rect = svg.getBoundingClientRect();
   125→    const [vw] = viewBox.split(" ").map(Number);
   126→    const sx = vw / rect.width;
   127→    const sy = sx;
   128→    const [, , , vh] = viewBox.split(" ").map(Number);
   129→    setViewBox(`${(vbStart.current.x - (e.clientX - dragStart.current.x) * sx).toFixed(1)} ${(vbStart.current.y - (e.clientY - dragStart.current.y) * sy).toFixed(1)} ${vw} ${vh}`);
   130→  }, [isDragging, viewBox]);
   131→
   132→  const highlightNode = selectedNationId || hoveredNode;
   133→
   134→  return (
   135→    <div className="relative w-full h-full min-h-[400px] bg-[#070b14] rounded-lg overflow-hidden border border-slate-800/50">
   136→      <svg className="absolute inset-0 w-full h-full opacity-[0.04] pointer-events-none">
   137→        <defs><pattern id="grid" width="5" height="5" patternUnits="userSpaceOnUse"><path d="M 5 0 L 0 0 0 5" fill="none" stroke="#06b6d4" strokeWidth="0.15" /></pattern></defs>
   138→        <rect width="100%" height="100%" fill="url(#grid)" />
   139→      </svg>
   140→      <div className="absolute top-3 left-4 z-10 flex items-center gap-2">
   141→        <div className="w-2 h-2 rounded-full bg-cyan-400 animate-pulse" />
   142→        <span className="text-xs font-mono text-cyan-400/70 uppercase tracking-wider">Grafo Mundial — {nations.length} nodos, {edges.length} aristas</span>
   143→      </div>
   144→      <div className="absolute top-3 right-3 z-10 flex flex-col gap-1.5 bg-black/40 backdrop-blur-sm rounded px-3 py-2 border border-slate-800/30">
   145→        {(["core", "semi", "periphery"] as const).map((c) => (
   146→          <div key={c} className="flex items-center gap-2">
   147→            <div className="w-2.5 h-2.5 rounded-full" style={{ background: classColor(c) }} />
   148→            <span className="text-[10px] text-slate-400">{classLabel(c)}</span>
   149→          </div>
   150→        ))}
   151→      </div>
   152→
   153→      <svg ref={svgRef} viewBox={viewBox} className="w-full h-full cursor-grab active:cursor-grabbing"
   154→        onWheel={handleWheel} onMouseDown={handleMouseDown} onMouseMove={handleMouseMove}
   155→        onMouseUp={() => setIsDragging(false)} onMouseLeave={() => setIsDragging(false)}
   156→        onClick={(e) => { if (!(e.target as Element).closest(".gn")) setSelectedNation(null); }}
   157→        style={{ touchAction: "none" }}>
   158→        <defs>
   159→          <filter id="gc" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="2" result="b" /><feFlood floodColor="#06b6d4" floodOpacity="0.6" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   160→          <filter id="gs" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="1.5" result="b" /><feFlood floodColor="#f59e0b" floodOpacity="0.5" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   161→          <filter id="gk" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="3" result="b" /><feFlood floodColor="#06b6d4" floodOpacity="0.8" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   162→        </defs>
   163→
   164→        {/* Edges */}
   165→        {edges.map((e, i) => {
   166→          const from = nations.find((n) => n.id === e.from);
   167→          const to = nations.find((n) => n.id === e.to);
   168→          if (!from || !to) return null;
   169→          const isHi = highlightNode ? (e.from === highlightNode || e.to === highlightNode) : false;
   170→          const op = highlightNode ? (isHi ? 0.7 : 0.05) : 0.15 + (e.volume / maxVolume) * 0.25;
   171→          const sc = isHi ? (e.type === "raw" ? "#10b981" : e.type === "manufactured" ? "#06b6d4" : "#a855f7") : "#334155";
   172→          return (
   173→            <line key={i} x1={from.x} y1={from.y} x2={to.x} y2={to.y}
   174→              stroke={sc} strokeWidth={isHi ? Math.min(3, e.volume / 100) * 1.5 : Math.min(3, e.volume / 100)}
   175→              opacity={op} strokeLinecap="round">
   176→              {isHi && <animate attributeName="stroke-dasharray" values="2 6;6 2;2 6" dur="2s" repeatCount="indefinite" />}
   177→            </line>
   178→          );
   179→        })}
   180→
   181→        {/* Nodes */}
   182→        {nations.map((n) => {
   183→          const r = Math.max(3, 3 + (n.gdp / 4.5) * 7);
   184→          const isSel = n.id === selectedNationId;
   185→          const isHov = n.id === hoveredNode;
   186→          const isConn = highlightNode && edges.some((e) => (e.from === highlightNode && e.to === n.id) || (e.to === highlightNode && e.from === n.id));
   187→          const dimmed = highlightNode && !isSel && !isConn && n.id !== highlightNode;
   188→          const color = classColor(n.worldClass);
   189→          return (
   190→            <g key={n.id} className="gn" transform={`translate(${n.x},${n.y})`} opacity={dimmed ? 0.15 : 1}
   191→              style={{ cursor: "pointer" }}
   192→              onClick={(ev) => {
   193→                ev.stopPropagation();
   194→                if (activeTab === "actions" && n.id !== "gb") {
   195→                  setActionTarget(isSel ? null : n.id);
   196→                } else {
   197→                  setSelectedNation(isSel ? null : n.id);
   198→                }
   199→              }}
   200→              onMouseEnter={() => setHoveredNode(n.id)} onMouseLeave={() => setHoveredNode(null)}>
   201→              {isSel && <circle r={r + 4} fill="none" stroke={color} strokeWidth="1.5" opacity="0.4"><animate attributeName="r" values={`${r + 3};${r + 6};${r + 3}`} dur="2s" repeatCount="indefinite" /><animate attributeName="opacity" values="0.4;0.15;0.4" dur="2s" repeatCount="indefinite" /></circle>}
   202→              {n.id === actionTargetId && (
   203→                <circle r={r + 3} fill="none" stroke="#ef4444" strokeWidth="1.2" strokeDasharray="3 2" opacity="0.8">
   204→                  <animateTransform attributeName="transform" type="rotate" from="0" to="360" dur="6s" repeatCount="indefinite" />
   205→                </circle>
   206→              )}
   207→              <circle r={r} fill={color} opacity={isSel || isHov ? 1 : 0.75}
   208→                filter={isSel ? "url(#gk)" : n.worldClass === "core" ? "url(#gc)" : n.worldClass === "semi" ? "url(#gs)" : undefined} />
   209→              <circle r={r * 0.4} fill="white" opacity={isSel ? 0.6 : 0.3} />
   210→              <text y={r + 3} textAnchor="middle" fill={dimmed ? "#475569" : isSel ? "#e2e8f0" : "#94a3b8"} fontSize="1.8" fontFamily="monospace" fontWeight={isSel ? 600 : 400}>
   211→                {n.flag} {n.name.length > 12 ? n.name.slice(0, 10) + "…" : n.name}
   212→              </text>
   213→              {isHov && !isSel && (
   214→                <g transform={`translate(0,${-r - 6})`}>
   215→                  <rect x={-30} y={-14} width={60} height={14} rx={3} fill="rgba(15,23,42,0.9)" stroke={color} strokeWidth={0.5} />
   216→                  <text textAnchor="middle" y={-4} fill="#e2e8f0" fontSize="2.2" fontFamily="monospace" fontWeight={600}>GDP: {n.gdp.toFixed(1)}B</text>
   217→                </g>
   218→              )}
   219→            </g>
   220→          );
   221→        })}
   222→      </svg>
   223→      <div className="absolute bottom-3 left-4 text-[10px] text-slate-600 font-mono">Scroll: zoom · Drag: pan · Click: select</div>
   224→    </div>
   225→  );
   226→}
