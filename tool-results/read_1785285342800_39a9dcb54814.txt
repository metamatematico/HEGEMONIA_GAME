    76→// ─── GRAPH VIEW ───
    77→function GraphView() {
    78→  const svgRef = useRef<SVGSVGElement>(null);
    79→  const [hoveredNode, setHoveredNode] = useState<string | null>(null);
    80→  const [viewBox, setViewBox] = useState("-5 -5 110 90");
    81→  const [isDragging, setIsDragging] = useState(false);
    82→  const dragStart = useRef({ x: 0, y: 0 });
    83→  const vbStart = useRef({ x: -5, y: -5 });
    84→
    85→  const nations = useHegemoniaStore((s) => s.nations);
    86→  const edges = useHegemoniaStore((s) => s.edges);
    87→  const selectedNationId = useHegemoniaStore((s) => s.selectedNationId);
    88→  const setSelectedNation = useHegemoniaStore((s) => s.setSelectedNation);
    89→  const activeTab = useHegemoniaStore((s) => s.activeTab);
    90→  const actionTargetId = useHegemoniaStore((s) => s.actionTargetId);
    91→  const setActionTarget = useHegemoniaStore((s) => s.setActionTarget);
    92→
    93→  const maxVolume = useMemo(() => Math.max(1, ...edges.map((e) => e.volume)), [edges]);
    94→
    95→  const handleWheel = useCallback((e: React.WheelEvent) => {
    96→    e.preventDefault();
    97→    const [vx, vy, vw, vh] = viewBox.split(" ").map(Number);
    98→    const svg = svgRef.current;
    99→    if (!svg) return;
   100→    const rect = svg.getBoundingClientRect();
   101→    const mx = ((e.clientX - rect.left) / rect.width) * vw + vx;
   102→    const my = ((e.clientY - rect.top) / rect.height) * vh + vy;
   103→    const f = e.deltaY > 0 ? 1.1 : 0.9;
   104→    const nvw = Math.max(30, Math.min(130, vw * f));
   105→    const nvh = nvw * (vh / vw);
   106→    const r = nvw / vw;
   107→    setViewBox(`${(mx - (mx - vx) * r).toFixed(1)} ${(my - (my - vy) * r).toFixed(1)} ${nvw.toFixed(1)} ${nvh.toFixed(1)}`);
   108→  }, [viewBox]);
   109→
   110→  const handleMouseDown = useCallback((e: React.MouseEvent) => {
   111→    if ((e.target as Element).closest(".gn")) return;
   112→    setIsDragging(true);
   113→    dragStart.current = { x: e.clientX, y: e.clientY };
   114→    const [vx, vy] = viewBox.split(" ").map(Number);
   115→    vbStart.current = { x: vx, y: vy };
   116→  }, [viewBox]);
   117→
   118→  const handleMouseMove = useCallback((e: React.MouseEvent) => {
   119→    if (!isDragging) return;
   120→    const svg = svgRef.current;
   121→    if (!svg) return;
   122→    const rect = svg.getBoundingClientRect();
   123→    const [vw] = viewBox.split(" ").map(Number);
   124→    const sx = vw / rect.width;
   125→    const sy = sx;
   126→    const [, , , vh] = viewBox.split(" ").map(Number);
   127→    setViewBox(`${(vbStart.current.x - (e.clientX - dragStart.current.x) * sx).toFixed(1)} ${(vbStart.current.y - (e.clientY - dragStart.current.y) * sy).toFixed(1)} ${vw} ${vh}`);
   128→  }, [isDragging, viewBox]);
   129→
   130→  const highlightNode = selectedNationId || hoveredNode;
   131→
   132→  return (
   133→    <div className="relative w-full h-full min-h-[400px] bg-[#070b14] rounded-lg overflow-hidden border border-slate-800/50">
   134→      <svg className="absolute inset-0 w-full h-full opacity-[0.04] pointer-events-none">
   135→        <defs><pattern id="grid" width="5" height="5" patternUnits="userSpaceOnUse"><path d="M 5 0 L 0 0 0 5" fill="none" stroke="#06b6d4" strokeWidth="0.15" /></pattern></defs>
   136→        <rect width="100%" height="100%" fill="url(#grid)" />
   137→      </svg>
   138→      <div className="absolute top-3 left-4 z-10 flex items-center gap-2">
   139→        <div className="w-2 h-2 rounded-full bg-cyan-400 animate-pulse" />
   140→        <span className="text-xs font-mono text-cyan-400/70 uppercase tracking-wider">Grafo Mundial — {nations.length} nodos, {edges.length} aristas</span>
   141→      </div>
   142→      <div className="absolute top-3 right-3 z-10 flex flex-col gap-1.5 bg-black/40 backdrop-blur-sm rounded px-3 py-2 border border-slate-800/30">
   143→        {(["core", "semi", "periphery"] as const).map((c) => (
   144→          <div key={c} className="flex items-center gap-2">
   145→            <div className="w-2.5 h-2.5 rounded-full" style={{ background: classColor(c) }} />
   146→            <span className="text-[10px] text-slate-400">{classLabel(c)}</span>
   147→          </div>
   148→        ))}
   149→      </div>
   150→
   151→      <svg ref={svgRef} viewBox={viewBox} className="w-full h-full cursor-grab active:cursor-grabbing"
   152→        onWheel={handleWheel} onMouseDown={handleMouseDown} onMouseMove={handleMouseMove}
   153→        onMouseUp={() => setIsDragging(false)} onMouseLeave={() => setIsDragging(false)}
   154→        onClick={(e) => { if (!(e.target as Element).closest(".gn")) setSelectedNation(null); }}
   155→        style={{ touchAction: "none" }}>
   156→        <defs>
   157→          <filter id="gc" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="2" result="b" /><feFlood floodColor="#06b6d4" floodOpacity="0.6" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   158→          <filter id="gs" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="1.5" result="b" /><feFlood floodColor="#f59e0b" floodOpacity="0.5" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   159→          <filter id="gk" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="3" result="b" /><feFlood floodColor="#06b6d4" floodOpacity="0.8" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   160→        </defs>
   161→
   162→        {/* Edges */}
   163→        {edges.map((e, i) => {
   164→          const from = nations.find((n) => n.id === e.from);
   165→          const to = nations.find((n) => n.id === e.to);
   166→          if (!from || !to) return null;
   167→          const isHi = highlightNode ? (e.from === highlightNode || e.to === highlightNode) : false;
   168→          const op = highlightNode ? (isHi ? 0.7 : 0.05) : 0.15 + (e.volume / maxVolume) * 0.25;
   169→          const sc = isHi ? (e.type === "raw" ? "#10b981" : e.type === "manufactured" ? "#06b6d4" : "#a855f7") : "#334155";
   170→          return (
   171→            <line key={i} x1={from.x} y1={from.y} x2={to.x} y2={to.y}
   172→              stroke={sc} strokeWidth={isHi ? Math.min(3, e.volume / 100) * 1.5 : Math.min(3, e.volume / 100)}
   173→              opacity={op} strokeLinecap="round">
   174→              {isHi && <animate attributeName="stroke-dasharray" values="2 6;6 2;2 6" dur="2s" repeatCount="indefinite" />}
   175→            </line>
   176→          );
   177→        })}
   178→
   179→        {/* Nodes */}
   180→        {nations.map((n) => {
   181→          const r = Math.max(3, 3 + (n.gdp / 4.5) * 7);
   182→          const isSel = n.id === selectedNationId;
   183→          const isHov = n.id === hoveredNode;
   184→          const isConn = highlightNode && edges.some((e) => (e.from === highlightNode && e.to === n.id) || (e.to === highlightNode && e.from === n.id));
   185→          const dimmed = highlightNode && !isSel && !isConn && n.id !== highlightNode;
   186→          const color = classColor(n.worldClass);
   187→          return (
   188→            <g key={n.id} className="gn" transform={`translate(${n.x},${n.y})`} opacity={dimmed ? 0.15 : 1}
   189→              style={{ cursor: "pointer" }}
   190→              onClick={(ev) => {
   191→                ev.stopPropagation();
   192→                if (activeTab === "actions" && n.id !== "gb") {
   193→                  setActionTarget(isSel ? null : n.id);
   194→                } else {
   195→                  setSelectedNation(isSel ? null : n.id);
   196→                }
   197→              }}
   198→              onMouseEnter={() => setHoveredNode(n.id)} onMouseLeave={() => setHoveredNode(null)}>
   199→              {isSel && <circle r={r + 4} fill="none" stroke={color} strokeWidth="1.5" opacity="0.4"><animate attributeName="r" values={`${r + 3};${r + 6};${r + 3}`} dur="2s" repeatCount="indefinite" /><animate attributeName="opacity" values="0.4;0.15;0.4" dur="2s" repeatCount="indefinite" /></circle>}
   200→              {n.id === actionTargetId && (
   201→                <circle r={r + 3} fill="none" stroke="#ef4444" strokeWidth="1.2" strokeDasharray="3 2" opacity="0.8">
   202→                  <animateTransform attributeName="transform" type="rotate" from="0" to="360" dur="6s" repeatCount="indefinite" />
   203→                </circle>
   204→              )}
   205→              <circle r={r} fill={color} opacity={isSel || isHov ? 1 : 0.75}
   206→                filter={isSel ? "url(#gk)" : n.worldClass === "core" ? "url(#gc)" : n.worldClass === "semi" ? "url(#gs)" : undefined} />
   207→              <circle r={r * 0.4} fill="white" opacity={isSel ? 0.6 : 0.3} />
   208→              <text y={r + 3} textAnchor="middle" fill={dimmed ? "#475569" : isSel ? "#e2e8f0" : "#94a3b8"} fontSize="1.8" fontFamily="monospace" fontWeight={isSel ? 600 : 400}>
   209→                {n.flag} {n.name.length > 12 ? n.name.slice(0, 10) + "…" : n.name}
   210→              </text>
   211→              {isHov && !isSel && (
   212→                <g transform={`translate(0,${-r - 6})`}>
   213→                  <rect x={-30} y={-14} width={60} height={14} rx={3} fill="rgba(15,23,42,0.9)" stroke={color} strokeWidth={0.5} />
   214→                  <text textAnchor="middle" y={-4} fill="#e2e8f0" fontSize="2.2" fontFamily="monospace" fontWeight={600}>GDP: {n.gdp.toFixed(1)}B</text>
   215→                </g>
   216→              )}
   217→            </g>
   218→          );
   219→        })}
   220→      </svg>
   221→      <div className="absolute bottom-3 left-4 text-[10px] text-slate-600 font-mono">Scroll: zoom · Drag: pan · Click: select</div>
   222→    </div>
   223→  );
   224→}
