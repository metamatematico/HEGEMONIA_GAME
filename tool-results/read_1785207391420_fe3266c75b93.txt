    74→    74→function GraphView() {
    75→    75→  const svgRef = useRef<SVGSVGElement>(null);
    76→    76→  const [hoveredNode, setHoveredNode] = useState<string | null>(null);
    77→    77→  const [viewBox, setViewBox] = useState("-5 -5 110 90");
    78→    78→  const [isDragging, setIsDragging] = useState(false);
    79→    79→  const dragStart = useRef({ x: 0, y: 0 });
    80→    80→  const vbStart = useRef({ x: -5, y: -5 });
    81→    81→
    82→    82→  const nations = useHegemoniaStore((s) => s.nations);
    83→    83→  const edges = useHegemoniaStore((s) => s.edges);
    84→    84→  const selectedNationId = useHegemoniaStore((s) => s.selectedNationId);
    85→    85→  const setSelectedNation = useHegemoniaStore((s) => s.setSelectedNation);
    86→    86→
    87→    87→  const maxVolume = useMemo(() => Math.max(1, ...edges.map((e) => e.volume)), [edges]);
    88→    88→
    89→    89→  const handleWheel = useCallback((e: React.WheelEvent) => {
    90→    90→    e.preventDefault();
    91→    91→    const [vx, vy, vw, vh] = viewBox.split(" ").map(Number);
    92→    92→    const svg = svgRef.current;
    93→    93→    if (!svg) return;
    94→    94→    const rect = svg.getBoundingClientRect();
    95→    95→    const mx = ((e.clientX - rect.left) / rect.width) * vw + vx;
    96→    96→    const my = ((e.clientY - rect.top) / rect.height) * vh + vy;
    97→    97→    const f = e.deltaY > 0 ? 1.1 : 0.9;
    98→    98→    const nvw = Math.max(30, Math.min(130, vw * f));
    99→    99→    const nvh = nvw * (vh / vw);
   100→   100→    const r = nvw / vw;
   101→   101→    setViewBox(`${(mx - (mx - vx) * r).toFixed(1)} ${(my - (my - vy) * r).toFixed(1)} ${nvw.toFixed(1)} ${nvh.toFixed(1)}`);
   102→   102→  }, [viewBox]);
   103→   103→
   104→   104→  const handleMouseDown = useCallback((e: React.MouseEvent) => {
   105→   105→    if ((e.target as Element).closest(".gn")) return;
   106→   106→    setIsDragging(true);
   107→   107→    dragStart.current = { x: e.clientX, y: e.clientY };
   108→   108→    const [vx, vy] = viewBox.split(" ").map(Number);
   109→   109→    vbStart.current = { x: vx, y: vy };
   110→   110→  }, [viewBox]);
   111→   111→
   112→   112→  const handleMouseMove = useCallback((e: React.MouseEvent) => {
   113→   113→    if (!isDragging) return;
   114→   114→    const svg = svgRef.current;
   115→   115→    if (!svg) return;
   116→   116→    const rect = svg.getBoundingClientRect();
   117→   117→    const [vw] = viewBox.split(" ").map(Number);
   118→   118→    const sx = vw / rect.width;
   119→   119→    const sy = sx;
   120→   120→    const [, , , vh] = viewBox.split(" ").map(Number);
   121→   121→    setViewBox(`${(vbStart.current.x - (e.clientX - dragStart.current.x) * sx).toFixed(1)} ${(vbStart.current.y - (e.clientY - dragStart.current.y) * sy).toFixed(1)} ${vw} ${vh}`);
   122→   122→  }, [isDragging, viewBox]);
   123→   123→
   124→   124→  const highlightNode = selectedNationId || hoveredNode;
   125→   125→
   126→   126→  return (
   127→   127→    <div className="relative w-full h-full min-h-[400px] bg-[#070b14] rounded-lg overflow-hidden border border-slate-800/50">
   128→   128→      <svg className="absolute inset-0 w-full h-full opacity-[0.04] pointer-events-none">
   129→   129→        <defs><pattern id="grid" width="5" height="5" patternUnits="userSpaceOnUse"><path d="M 5 0 L 0 0 0 5" fill="none" stroke="#06b6d4" strokeWidth="0.15" /></pattern></defs>
   130→   130→        <rect width="100%" height="100%" fill="url(#grid)" />
   131→   131→      </svg>
   132→   132→      <div className="absolute top-3 left-4 z-10 flex items-center gap-2">
   133→   133→        <div className="w-2 h-2 rounded-full bg-cyan-400 animate-pulse" />
   134→   134→        <span className="text-xs font-mono text-cyan-400/70 uppercase tracking-wider">Grafo Mundial — {nations.length} nodos, {edges.length} aristas</span>
   135→   135→      </div>
   136→   136→      <div className="absolute top-3 right-3 z-10 flex flex-col gap-1.5 bg-black/40 backdrop-blur-sm rounded px-3 py-2 border border-slate-800/30">
   137→   137→        {(["core", "semi", "periphery"] as const).map((c) => (
   138→   138→          <div key={c} className="flex items-center gap-2">
   139→   139→            <div className="w-2.5 h-2.5 rounded-full" style={{ background: classColor(c) }} />
   140→   140→            <span className="text-[10px] text-slate-400">{classLabel(c)}</span>
   141→   141→          </div>
   142→   142→        ))}
   143→   143→      </div>
   144→   144→
   145→   145→      <svg ref={svgRef} viewBox={viewBox} className="w-full h-full cursor-grab active:cursor-grabbing"
   146→   146→        onWheel={handleWheel} onMouseDown={handleMouseDown} onMouseMove={handleMouseMove}
   147→   147→        onMouseUp={() => setIsDragging(false)} onMouseLeave={() => setIsDragging(false)}
   148→   148→        onClick={(e) => { if (!(e.target as Element).closest(".gn")) setSelectedNation(null); }}
   149→   149→        style={{ touchAction: "none" }}>
   150→   150→        <defs>
   151→   151→          <filter id="gc" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="2" result="b" /><feFlood floodColor="#06b6d4" floodOpacity="0.6" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   152→   152→          <filter id="gs" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="1.5" result="b" /><feFlood floodColor="#f59e0b" floodOpacity="0.5" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   153→   153→          <filter id="gk" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="3" result="b" /><feFlood floodColor="#06b6d4" floodOpacity="0.8" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   154→   154→        </defs>
   155→   155→
   156→   156→        {/* Edges */}
   157→   157→        {edges.map((e, i) => {
   158→   158→          const from = nations.find((n) => n.id === e.from);
   159→   159→          const to = nations.find((n) => n.id === e.to);
   160→   160→          if (!from || !to) return null;
   161→   161→          const isHi = highlightNode ? (e.from === highlightNode || e.to === highlightNode) : false;
   162→   162→          const op = highlightNode ? (isHi ? 0.7 : 0.05) : 0.15 + (e.volume / maxVolume) * 0.25;
   163→   163→          const sc = isHi ? (e.type === "raw" ? "#10b981" : e.type === "manufactured" ? "#06b6d4" : "#a855f7") : "#334155";
   164→   164→          return (
   165→   165→            <line key={i} x1={from.x} y1={from.y} x2={to.x} y2={to.y}
   166→   166→              stroke={sc} strokeWidth={isHi ? Math.min(3, e.volume / 100) * 1.5 : Math.min(3, e.volume / 100)}
   167→   167→              opacity={op} strokeLinecap="round">
   168→   168→              {isHi && <animate attributeName="stroke-dasharray" values="2 6;6 2;2 6" dur="2s" repeatCount="indefinite" />}
   169→   169→            </line>
   170→   170→          );
   171→   171→        })}
   172→   172→
   173→   173→        {/* Nodes */}
   174→   174→        {nations.map((n) => {
   175→   175→          const r = Math.max(3, 3 + (n.gdp / 4.5) * 7);
   176→   176→          const isSel = n.id === selectedNationId;
   177→   177→          const isHov = n.id === hoveredNode;
   178→   178→          const isConn = highlightNode && edges.some((e) => (e.from === highlightNode && e.to === n.id) || (e.to === highlightNode && e.from === n.id));
   179→   179→          const dimmed = highlightNode && !isSel && !isConn && n.id !== highlightNode;
   180→   180→          const color = classColor(n.worldClass);
   181→   181→          return (
   182→   182→            <g key={n.id} className="gn" transform={`translate(${n.x},${n.y})`} opacity={dimmed ? 0.15 : 1}
   183→   183→              style={{ cursor: "pointer" }}
   184→   184→              onClick={(ev) => { ev.stopPropagation(); setSelectedNation(isSel ? null : n.id); }}
   185→   185→              onMouseEnter={() => setHoveredNode(n.id)} onMouseLeave={() => setHoveredNode(null)}>
   186→   186→              {isSel && <circle r={r + 4} fill="none" stroke={color} strokeWidth="1.5" opacity="0.4"><animate attributeName="r" values={`${r + 3};${r + 6};${r + 3}`} dur="2s" repeatCount="indefinite" /><animate attributeName="opacity" values="0.4;0.15;0.4" dur="2s" repeatCount="indefinite" /></circle>}
   187→   187→              <circle r={r} fill={color} opacity={isSel || isHov ? 1 : 0.75}
   188→   188→                filter={isSel ? "url(#gk)" : n.worldClass === "core" ? "url(#gc)" : n.worldClass === "semi" ? "url(#gs)" : undefined} />
   189→   189→              <circle r={r * 0.4} fill="white" opacity={isSel ? 0.6 : 0.3} />
   190→   190→              <text y={r + 3} textAnchor="middle" fill={dimmed ? "#475569" : isSel ? "#e2e8f0" : "#94a3b8"} fontSize="1.8" fontFamily="monospace" fontWeight={isSel ? 600 : 400}>
   191→   191→                {n.flag} {n.name.length > 12 ? n.name.slice(0, 10) + "…" : n.name}
   192→   192→              </text>
   193→   193→              {isHov && !isSel && (
   194→   194→                <g transform={`translate(0,${-r - 6})`}>
   195→   195→                  <rect x={-30} y={-14} width={60} height={14} rx={3} fill="rgba(15,23,42,0.9)" stroke={color} strokeWidth={0.5} />
   196→   196→                  <text textAnchor="middle" y={-4} fill="#e2e8f0" fontSize="2.2" fontFamily="monospace" fontWeight={600}>GDP: {n.gdp.toFixed(1)}B</text>
   197→   197→                </g>
   198→   198→              )}
   199→   199→            </g>
   200→   200→          );
   201→   201→        })}
   202→   202→      </svg>
   203→   203→      <div className="absolute bottom-3 left-4 text-[10px] text-slate-600 font-mono">Scroll: zoom · Drag: pan · Click: select</div>
   204→   204→    </div>
   205→   205→  );
   206→   206→}
