    15→export interface IdeologyProfile {
    16→  /** Production efficiency multiplier */
    17→  productionMod: number;
    18→  /** Trade openness: 0 = autarky, 1 = free trade */
    19→  openness: number;
    20→  /** Preferred tariff rate */
    21→  idealTariff: number;
    22→  /** How much cultural pressure this ideology exerts on neighbors */
    23→  culturalPressure: number;
    24→  /** How susceptible to incoming ideological pressure */
    25→  susceptibility: number;
    26→  /** Military spending priority (0-1) */
    27→  militaryPriority: number;
    28→  /** Industrialization speed bonus */
    29→  industrialBonus: number;
    30→  /** Social mobility (how fast peasants → working/middle class) */
    31→  socialMobility: number;
    32→  /** Stability bonus (natural order/legitimacy) */
    33→  stabilityBonus: number;
    34→  /** Unrest generation (inherent dissatisfaction) */
    35→  unrestGeneration: number;
    36→  /** Trade type affinity: "raw" = extractive, "manufactured" = industrial, "luxury" = both */
    37→  exportAffinity: "raw" | "manufactured" | "luxury";
    38→  /** Description of graph mechanic */
    39→  graphMechanic: string;
    40→  /** Key modifier: how this ideology changes edge weights */
    41→  edgeWeightModifier: (volume: number, fromClass: string, toClass: string) => number;
    42→  /** Unique per-turn ideological action */
    43→  ideologicalEffect: (nation: NationNode, nations: NationNode[], edges: TradeEdge[]) => IdeologyEffect[];
    44→}
    45→
    46→export interface IdeologyEffect {
    47→  type: "trade" | "war" | "revolution" | "crisis" | "diplomacy" | "ideology";
    48→  text: string;
    49→  nationId: string;
    50→}

    54→function mercantilismEdgeMod(volume: number, fromClass: string, _toClass: string): number {
    55→  // Mercantilism: boosts exports to periphery, weakens imports from core competitors
    56→  if (fromClass === "core" || fromClass === "semi") return volume * 1.15;
    57→  return volume * 0.95;
    58→}
    59→
    60→function mercantilismEffect(
    61→  nation: NationNode, nations: NationNode[], edges: TradeEdge[]
    62→): IdeologyEffect[] {
    63→  const effects: IdeologyEffect[] = [];
    64→
    65→  // Mercantilism: actively seeks trade surplus — raise tariffs on imports
    66→  for (const e of edges) {
    67→    if (e.to === nation.id && e.type === "manufactured") {
    68→      e.tariffRate = Math.min(0.5, e.tariffRate + 0.02);
    69→    }
    70→    if (e.from === nation.id && e.type === "raw") {
    71→      e.tariffRate = Math.max(0.01, e.tariffRate - 0.01);
    72→    }
    73→  }
    74→
    75→  // Accumulation: mercantilist nations hoard resources
    76→  if (nation.resources.coal < 50) nation.resources.coal += 0.3;
    77→  if (nation.resources.iron < 50) nation.resources.iron += 0.2;
    78→
    79→  // Event: colonial extraction
    80→  if (Math.random() < 0.03) {
    81→    const colonies = nations.filter(
    82→      (n) => n.worldClass === "periphery" && n.id !== nation.id
    83→    );
    84→    if (colonies.length > 0) {
    85→      const colony = colonies[Math.floor(Math.random() * colonies.length)];
    86→      // Extract resources from colony
    87→      colony.resources.grain = Math.max(5, colony.resources.grain - 1);
    88→      colony.resources.cotton = Math.max(5, colony.resources.cotton - 1);
    89→      nation.gdp += 0.01;
    90→      effects.push({
    91→        type: "trade",
    92→        text: `${nation.name}: Extracción colonial de ${colony.name}`,
    93→        nationId: nation.id,
    94→      });
    95→    }
    96→  }
    97→
    98→  return effects;
    99→}

   101→function liberalismEdgeMod(volume: number, _fromClass: string, toClass: string): number {
   102→  // Liberalism: free trade benefits everyone, especially core trading with periphery
   103→  if (toClass === "periphery") return volume * 1.1;
   104→  return volume * 1.05;

   107→function liberalismEffect(
   108→  nation: NationNode, nations: NationNode[], edges: TradeEdge[]
   109→): IdeologyEffect[] {
   110→  const effects: IdeologyEffect[] = [];
   111→
   112→  // Liberalism: lower all tariffs toward free trade
   113→  for (const e of edges) {
   114→    if (e.from === nation.id || e.to === nation.id) {
   115→      e.tariffRate = Math.max(0.01, e.tariffRate * 0.98);
   116→    }
   117→  }
   118→
   119→  // Market innovation: faster industrialization through competition
   120→  nation.industrialization = Math.min(100, nation.industrialization + 0.1);
   121→
   122→  // Event: trade agreement proposal
   123→  if (Math.random() < 0.04) {
   124→    const potentialPartners = nations.filter(
   125→      (n) => n.id !== nation.id && n.primaryIdeology !== "absolutism"
   126→    );
   127→    if (potentialPartners.length > 0) {
   128→      const partner = potentialPartners[Math.floor(Math.random() * potentialPartners.length)];
   129→      // Boost existing trade or note the attempt
   130→      for (const e of edges) {
   131→        if (
   132→          (e.from === nation.id && e.to === partner.id) ||
   133→          (e.to === nation.id && e.from === partner.id)
   134→        ) {
   135→          e.volume = Math.min(1000, e.volume * 1.1);
   136→          effects.push({
   137→            type: "trade",
   138→            text: `${nation.name}: Acuerdo de libre comercio con ${partner.name}`,
   139→            nationId: nation.id,
   140→          });
   141→          return effects;
   142→        }
   143→      }
   144→    }
   145→  }
   146→
   147→  return effects;
   148→}

   150→function marxismEdgeMod(volume: number, fromClass: string, toClass: string): number {
   151→  // Marxism: reduces exploitation flows (core ← raw from periphery)
   152→  // Strengths solidarity between periphery nations
   153→  if (fromClass === "periphery" && toClass === "periphery") return volume * 1.3;
   154→  if (fromClass === "periphery" && toClass === "core") return volume * 0.85;
   155→  return volume * 0.9;
   156→}

   158→function marxismEffect(
   159→  nation: NationNode, nations: NationNode[], edges: TradeEdge[]
   160→): IdeologyEffect[] {
   161→  const effects: IdeologyEffect[] = [];
   162→
   163→  // Marxism: redistribute wealth, boost working class
   164→  nation.socialClasses.working = Math.min(55, nation.socialClasses.working + 0.3);
   165→  nation.socialClasses.elite = Math.max(2, nation.socialClasses.elite - 0.1);
   166→  nation.stability = Math.min(95, nation.stability + 2);
   167→  nation.unrest = Math.max(0, nation.unrest - 5);
   168→
   169→  // Socialist industrialization (collective ownership)
   170→  nation.industrialization = Math.min(100, nation.industrialization + 0.15);
   171→
   172→  // Event: inspire revolution in neighbors
   173→  if (nation.culturalPower > 30 && Math.random() < 0.05) {
   174→    const neighbors = new Set<string>();
   175→    for (const e of edges) {
   176→      if (e.from === nation.id) neighbors.add(e.to);
   177→      if (e.to === nation.id) neighbors.add(e.from);
   178→    }
   179→    const affected = nations.filter(
   180→      (n) => neighbors.has(n.id) && n.worldClass === "periphery" && n.unrest > 40
   181→    );
   182→    if (affected.length > 0) {
   183→      const target = affected[Math.floor(Math.random() * affected.length)];
   184→      target.unrest = Math.min(100, target.unrest + 10);
   185→      target.stability = Math.max(5, target.stability - 5);
   186→      effects.push({
   187→        type: "revolution",
   188→        text: `${nation.name}: Propaganda socialista inflama descontento en ${target.name}`,
   189→        nationId: nation.id,
   190→      });
   191→    }
   192→  }
   193→
   194→  return effects;
   195→}

   278→function absolutismEffect(
   279→  nation: NationNode, nations: NationNode[], edges: TradeEdge[]
   280→): IdeologyEffect[] {
   281→  const effects: IdeologyEffect[] = [];
   282→
   283→  // Absolutism: centralize power, suppress dissent
   284→  nation.unrest = Math.max(0, nation.unrest - 8);
   285→  nation.stability = Math.min(95, nation.stability + 3);
   286→  nation.militaryPower = Math.min(100, nation.militaryPower + 0.5);
   287→
   288→  // Suppress social mobility
   289→  nation.socialClasses.elite = Math.min(20, nation.socialClasses.elite + 0.1);
   290→  nation.socialClasses.peasant = Math.min(90, nation.socialClasses.peasant + 0.1);
   291→
   292→  // Event: autocratic decree
   293→  if (Math.random() < 0.03) {
   294→    // Restrict trade to state-controlled routes
   295→    const nationsToDrop = nations.filter((n) => {
   296→      if (n.id === nation.id) return false;
   297→      return edges.some(
   298→        (e) =>
   299→          ((e.from === nation.id && e.to === n.id) ||
   300→           (e.to === nation.id && e.from === n.id)) &&
   301→          e.volume < 20
   302→      );
   303→    });
   304→
   305→    if (nationsToDrop.length > 0) {
   306→      const target = nationsToDrop[Math.floor(Math.random() * nationsToDrop.length)];
   307→      // Reduce weak trade routes
   308→      for (let i = edges.length - 1; i >= 0; i--) {
   309→        const e = edges[i];
   310→        if (
   311→          ((e.from === nation.id && e.to === target.id) ||
   312→           (e.to === nation.id && e.from === target.id)) &&
   313→          e.volume < 20
   314→        ) {
   315→          e.volume = Math.max(1, e.volume - 5);
   316→        }
   317→      }
   318→      effects.push({
   319→        type: "diplomacy",
   320→        text: `${nation.name}: Decreto absolutista restringe comercio con ${target.name}`,
   321→        nationId: nation.id,
   322→      });
   323→    }
   324→  }
   325→
   326→  return effects;
   327→}

   335→function imperialismEffect(
   336→  nation: NationNode, nations: NationNode[], edges: TradeEdge[]
   337→): IdeologyEffect[] {
   338→  const effects: IdeologyEffect[] = [];
   339→
   340→  // Imperialism: actively extract resources from periphery neighbors
   341→  for (const e of edges) {
   342→    if (e.from === nation.id) {
   343→      const toNation = nations.find((n) => n.id === e.to);
   344→      if (toNation && toNation.worldClass === "periphery") {
   345→        // Increase extraction volume
   346→        e.volume = Math.min(500, e.volume * 1.03);
   347→      }
   348→    }
   349→  }
   350→
   351→  // Military buildup for imperial power
   352→  nation.militaryPower = Math.min(100, nation.militaryPower + 1.0);
   353→
   354→  // Reduce colonial stability in periphery neighbors
   355→  const neighbors = new Set<string>();
   356→  for (const e of edges) {
   357→    if (e.from === nation.id) neighbors.add(e.to);
   358→    if (e.to === nation.id) neighbors.add(e.from);
   359→  }
   360→  for (const neighborId of neighbors) {
   361→    const neighbor = nations.find((n) => n.id === neighborId);
   362→    if (neighbor && neighbor.worldClass === "periphery") {
   363→      neighbor.stability = Math.max(5, neighbor.stability - 0.5);
   364→    }
   365→  }
   366→
   367→  // Event: colonial expansion / resource seizure
   368→  if (Math.random() < 0.03 && nation.militaryPower > 40) {
   369→    const colonies = nations.filter(
   370→      (n) => n.worldClass === "periphery" && n.id !== nation.id && neighbors.has(n.id)
   371→    );
   372→    if (colonies.length > 0) {
   373→      const colony = colonies[Math.floor(Math.random() * colonies.length)];
   374→      colony.resources.grain = Math.max(5, colony.resources.grain - 2);
   375→      colony.resources.coal = Math.max(5, colony.resources.coal - 1);
   376→      colony.stability = Math.max(5, colony.stability - 3);
   377→      nation.gdp += 0.02;
   378→      effects.push({
   379→        type: "trade",
   380→        text: `${nation.name}: Extracción imperialista de recursos en ${colony.name}`,
   381→        nationId: nation.id,
   382→      });
   383→    }
   384→  }
   385→
   386→  return effects;
   387→}

   553→function theocracyEffect(
   554→  nation: NationNode, nations: NationNode[], edges: TradeEdge[]
   555→): IdeologyEffect[] {
   556→  const effects: IdeologyEffect[] = [];
   557→
   558→  // Theocracy: massive stability from religious authority

   560→  nation.unrest = Math.max(0, nation.unrest - 4);
   561→

   563→  nation.culturalPower = Math.min(100, nation.culturalPower + 1.0);
   564→

   567→  nation.socialClasses.elite = Math.min(20, nation.socialClasses.elite + 0.05);
   568→
   569→  // Maintain moderate tariffs
   570→  for (const e of edges) {
   571→    if (e.from === nation.id || e.to === nation.id) {
   572→      e.tariffRate += (0.18 - e.tariffRate) * 0.04;
   573→    }
   574→  }
   575→
   576→  // Event: religious mission / cultural spread to neighbors

   582→    }
   583→    const targets = nations.filter(
   584→      (n) => neighbors.has(n.id) && n.id !== nation.id

   604→function syndicalismEdgeMod(volume: number, fromClass: string, toClass: string): number {
   605→  // Syndicalism: boosts trade between periphery nations (worker solidarity)
   606→  if (fromClass === "periphery" || toClass === "periphery") return volume * 1.15;
   607→  return volume * 1.0;

   912→export function applyIdeologicalEffects(
   913→  nations: NationNode[],
   914→  edges: TradeEdge[]
   915→): IdeologyEffect[] {
   916→  const allEffects: IdeologyEffect[] = [];
   917→
   918→  for (const nation of nations) {
   919→    const profile = IDEOLOGY_PROFILES[nation.primaryIdeology];
   920→
   921→    // Apply edge weight modifiers
   922→    for (const e of edges) {
   923→      if (e.from === nation.id) {
   924→        const toNation = nations.find((n) => n.id === e.to);
   925→        if (toNation) {
   926→          const modified = profile.edgeWeightModifier(e.volume, nation.worldClass, toNation.worldClass);
   927→          e.volume = Math.max(1, e.volume + (modified - e.volume) * 0.02); // gentle drift
   928→        }
   929→      }
   930→    }
   931→
   932→    // Apply per-turn ideological effects
   933→    const effects = profile.ideologicalEffect(nation, nations, edges);
   934→    allEffects.push(...effects);
   935→
   936→    // Apply passive bonuses
   937→    nation.stability = Math.min(
   938→      95,
   939→      Math.max(5, nation.stability + profile.stabilityBonus * 0.01)
   940→    );
   941→    nation.unrest = Math.min(
   942→      100,
   943→      Math.max(0, nation.unrest + profile.unrestGeneration * 0.5)
   944→    );
   945→  }

  1036→/**
  1037→ * Compute the dominant ideology in a nation's trade neighborhood.
  1038→ * Used for ideological pressure calculations.
  1039→ */
  1040→export function neighborhoodIdeologyPressure(
  1041→  nation: NationNode,
  1042→  nations: NationNode[],
  1043→  edges: TradeEdge[]
  1044→): Map<Ideology, number> {
  1045→  const pressure = new Map<Ideology, number>();
  1046→
  1047→  for (const e of edges) {
  1048→    const partnerId = e.from === nation.id ? e.to : e.from;
  1049→    if (partnerId === nation.id) continue;
  1050→
  1051→    const partner = nations.find((n) => n.id === partnerId);
