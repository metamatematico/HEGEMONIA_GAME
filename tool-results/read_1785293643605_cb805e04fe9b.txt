    85→    85→function GraphView() {
    86→    86→  const svgRef = useRef<SVGSVGElement>(null);
    87→    87→  const [hoveredNode, setHoveredNode] = useState<string | null>(null);
    88→    88→  const [viewBox, setViewBox] = useState("-5 -5 110 90");
    89→    89→  const [isDragging, setIsDragging] = useState(false);
    90→    90→  const dragStart = useRef({ x: 0, y: 0 });
    91→    91→  const vbStart = useRef({ x: -5, y: -5 });
    92→    92→
    93→    93→  const nations = useHegemoniaStore((s) => s.nations);
    94→    94→  const edges = useHegemoniaStore((s) => s.edges);
    95→    95→  const selectedNationId = useHegemoniaStore((s) => s.selectedNationId);
    96→    96→  const setSelectedNation = useHegemoniaStore((s) => s.setSelectedNation);
    97→    97→  const activeTab = useHegemoniaStore((s) => s.activeTab);
    98→    98→  const actionTargetId = useHegemoniaStore((s) => s.actionTargetId);
    99→    99→  const setActionTarget = useHegemoniaStore((s) => s.setActionTarget);
   100→   100→
   101→   101→  const maxVolume = useMemo(() => Math.max(1, ...edges.map((e) => e.volume)), [edges]);
   102→   102→
   103→   103→  // Zoom centered on cursor
   104→   104→  const handleWheel = useCallback((e: React.WheelEvent) => {
   105→   105→    e.preventDefault();
   106→   106→    const [vx, vy, vw, vh] = viewBox.split(" ").map(Number);
   107→   107→    const svg = svgRef.current;
   108→   108→    if (!svg) return;
   109→   109→    const rect = svg.getBoundingClientRect();
   110→   110→    const mx = ((e.clientX - rect.left) / rect.width) * vw + vx;
   111→   111→    const my = ((e.clientY - rect.top) / rect.height) * vh + vy;
   112→   112→    const f = e.deltaY > 0 ? 1.1 : 0.9;
   113→   113→    const nvw = Math.max(20, Math.min(150, vw * f));
   114→   114→    const nvh = nvw * (vh / vw);
   115→   115→    const r = nvw / vw;
   116→   116→    setViewBox(`${(mx - (mx - vx) * r).toFixed(1)} ${(my - (my - vy) * r).toFixed(1)} ${nvw.toFixed(1)} ${nvh.toFixed(1)}`);
   117→   117→  }, [viewBox]);
   118→   118→
   119→   119→  // Pan
   120→   120→  const handleMouseDown = useCallback((e: React.MouseEvent) => {
   121→   121→    if ((e.target as Element).closest(".gn")) return;
   122→   122→    setIsDragging(true);
   123→   123→    dragStart.current = { x: e.clientX, y: e.clientY };
   124→   124→    const [vx, vy] = viewBox.split(" ").map(Number);
   125→   125→    vbStart.current = { x: vx, y: vy };
   126→   126→  }, [viewBox]);
   127→   127→
   128→   128→  const handleMouseMove = useCallback((e: React.MouseEvent) => {
   129→   129→    if (!isDragging) return;
   130→   130→    const svg = svgRef.current;
   131→   131→    if (!svg) return;
   132→   132→    const rect = svg.getBoundingClientRect();
   133→   133→    const [vw, , , vh] = viewBox.split(" ").map(Number);
   134→   134→    const sx = vw / rect.width;
   135→   135→    const sy = sx;
   136→   136→    setViewBox(`${(vbStart.current.x - (e.clientX - dragStart.current.x) * sx).toFixed(1)} ${(vbStart.current.y - (e.clientY - dragStart.current.y) * sy).toFixed(1)} ${vw} ${vh}`);
   137→   137→  }, [isDragging, viewBox]);
   138→   138→
   139→   139→  // Reset view
   140→   140→  const resetView = useCallback(() => setViewBox("-5 -5 110 90"), []);
   141→   141→
   142→   142→  const highlightNode = selectedNationId || hoveredNode;
   143→   143→
   144→   144→  return (
   145→   145→    <div className="relative w-full h-full min-h-[300px] bg-[#070b14] rounded-lg overflow-hidden border border-slate-800/50">
   146→   146→      {/* Grid background */}
   147→   147→      <svg className="absolute inset-0 w-full h-full opacity-[0.04] pointer-events-none">
   148→   148→        <defs><pattern id="grid" width="5" height="5" patternUnits="userSpaceOnUse"><path d="M 5 0 L 0 0 0 5" fill="none" stroke="#06b6d4" strokeWidth="0.15" /></pattern></defs>
   149→   149→        <rect width="100%" height="100%" fill="url(#grid)" />
   150→   150→      </svg>
   151→   151→
   152→   152→      {/* Top bar */}
   153→   153→      <div className="absolute top-3 left-4 z-10 flex items-center gap-2">
   154→   154→        <div className="w-2 h-2 rounded-full bg-cyan-400 animate-pulse" />
   155→   155→        <span className="text-xs font-mono text-cyan-400/70 uppercase tracking-wider">Grafo Mundial — {nations.length} nodos, {edges.length} aristas</span>
   156→   156→      </div>
   157→   157→
   158→   158→      {/* Legend */}
   159→   159→      <div className="absolute top-3 right-3 z-10 flex flex-col gap-1.5 bg-black/40 backdrop-blur-sm rounded px-3 py-2 border border-slate-800/30">
   160→   160→        {(["core", "semi", "periphery"] as const).map((c) => (
   161→   161→          <div key={c} className="flex items-center gap-2">
   162→   162→            <div className="w-2.5 h-2.5 rounded-full" style={{ background: classColor(c) }} />
   163→   163→            <span className="text-[10px] text-slate-400">{classLabel(c)}</span>
   164→   164→          </div>
   165→   165→        ))}
   166→   166→      </div>
   167→   167→
   168→   168→      {/* Controls bar */}
   169→   169→      <div className="absolute bottom-3 right-3 z-10 flex gap-1.5">
   170→   170→        <button onClick={resetView}
   171→   171→          className="w-7 h-7 rounded bg-black/40 backdrop-blur-sm border border-slate-800/30 flex items-center justify-center hover:bg-slate-800/50 transition-colors"
   172→   172→          title="Resetear vista">
   173→   173→          <Maximize2 className="w-3 h-3 text-slate-400" />
   174→   174→        </button>
   175→   175→        <button onClick={() => setViewBox(vb => { const [x,y,w,h] = vb.split(" ").map(Number); return `${x.toFixed(1)} ${y.toFixed(1)} ${(w*0.7).toFixed(1)} ${(h*0.7).toFixed(1)}`; })}
   176→   176→          className="w-7 h-7 rounded bg-black/40 backdrop-blur-sm border border-slate-800/30 flex items-center justify-center hover:bg-slate-800/50 transition-colors text-[11px] text-slate-400 font-bold"
   177→   177→          title="Zoom +">
   178→   178→          +
   179→   179→        </button>
   180→   180→        <button onClick={() => setViewBox(vb => { const [x,y,w,h] = vb.split(" ").map(Number); return `${(x-w*0.15).toFixed(1)} ${(y-h*0.15).toFixed(1)} ${(w*1.3).toFixed(1)} ${(h*1.3).toFixed(1)}`; })}
   181→   181→          className="w-7 h-7 rounded bg-black/40 backdrop-blur-sm border border-slate-800/30 flex items-center justify-center hover:bg-slate-800/50 transition-colors text-[11px] text-slate-400 font-bold"
   182→   182→          title="Zoom -">
   183→   183→          −
   184→   184→        </button>
   185→   185→      </div>
   186→   186→
   187→   187→      {/* SVG Graph */}
   188→   188→      <svg ref={svgRef} viewBox={viewBox} className="w-full h-full cursor-grab active:cursor-grabbing"
   189→   189→        onWheel={handleWheel} onMouseDown={handleMouseDown} onMouseMove={handleMouseMove}
   190→   190→        onMouseUp={() => setIsDragging(false)} onMouseLeave={() => setIsDragging(false)}
   191→   191→        onClick={(e) => { if (!(e.target as Element).closest(".gn")) setSelectedNation(null); }}
   192→   192→        style={{ touchAction: "none" }}>
   193→   193→        <defs>
   194→   194→          <filter id="gc" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="2" result="b" /><feFlood floodColor="#06b6d4" floodOpacity="0.6" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   195→   195→          <filter id="gs" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="1.5" result="b" /><feFlood floodColor="#f59e0b" floodOpacity="0.5" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   196→   196→          <filter id="gk" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="3" result="b" /><feFlood floodColor="#06b6d4" floodOpacity="0.8" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   197→   197→        </defs>
   198→   198→
   199→   199→        {/* Edges */}
   200→   200→        {edges.map((e, i) => {
   201→   201→          const from = nations.find((n) => n.id === e.from);
   202→   202→          const to = nations.find((n) => n.id === e.to);
   203→   203→          if (!from || !to) return null;
   204→   204→          const isHi = highlightNode ? (e.from === highlightNode || e.to === highlightNode) : false;
   205→   205→          const op = highlightNode ? (isHi ? 0.7 : 0.05) : 0.15 + (e.volume / maxVolume) * 0.25;
   206→   206→          const sc = isHi ? (e.type === "raw" ? "#10b981" : e.type === "manufactured" ? "#06b6d4" : "#a855f7") : "#334155";

   208→   208→            <line key={i} x1={from.x} y1={from.y} x2={to.x} y2={to.y}
   209→   209→              stroke={sc} strokeWidth={isHi ? Math.min(3, e.volume / 100) * 1.5 : Math.min(3, e.volume / 100)}
   210→   210→              opacity={op} strokeLinecap="round">

   212→   212→            </line>
   213→   213→          );
   214→   214→        })}
   215→   215→
   216→   216→        {/* Nodes */}
   217→   217→        {nations.map((n) => {
   218→   218→          const r = Math.max(3, 3 + (n.gdp / 4.5) * 7);

   220→   220→          const isHov = n.id === hoveredNode;
   221→   221→          const isConn = highlightNode && edges.some((e) => (e.from === highlightNode && e.to === n.id) || (e.to === highlightNode && e.from === n.id));

   223→   223→          const color = classColor(n.worldClass);
   224→   224→          return (
   225→   225→            <g key={n.id} className="gn" transform={`translate(${n.x},${n.y})`} opacity={dimmed ? 0.15 : 1}
   226→   226→              style={{ cursor: "pointer" }}
   227→   227→              onClick={(ev) => {
   228→   228→                ev.stopPropagation();
   229→   229→                if (activeTab === "actions" && n.id !== "gb") {
   230→   230→                  setActionTarget(isSel ? null : n.id);
   231→   231→                } else {

   237→   237→              {n.id === actionTargetId && (
   238→   238→                <circle r={r + 3} fill="none" stroke="#ef4444" strokeWidth="1.2" strokeDasharray="3 2" opacity="0.8">
   239→   239→                  <animateTransform attributeName="transform" type="rotate" from="0" to="360" dur="6s" repeatCount="indefinite" />
   240→   240→                </circle>
   241→   241→              )}
   242→   242→              <circle r={r} fill={color} opacity={isSel || isHov ? 1 : 0.75}
   243→   243→                filter={isSel ? "url(#gk)" : n.worldClass === "core" ? "url(#gc)" : n.worldClass === "semi" ? "url(#gs)" : undefined} />
   244→   244→              <circle r={r * 0.4} fill="white" opacity={isSel ? 0.6 : 0.3} />
   245→   245→              <text y={r + 3} textAnchor="middle" fill={dimmed ? "#475569" : isSel ? "#e2e8f0" : "#94a3b8"} fontSize="1.8" fontFamily="monospace" fontWeight={isSel ? 600 : 400}>
   246→   246→                {n.flag} {n.name.length > 12 ? n.name.slice(0, 10) + "…" : n.name}
   247→   247→              </text>
   248→   248→              {isHov && !isSel && (
   249→   249→                <g transform={`translate(0,${-r - 6})`}>
   250→   250→                  <rect x={-30} y={-14} width={60} height={14} rx={3} fill="rgba(15,23,42,0.9)" stroke={color} strokeWidth={0.5} />
   251→   251→                  <text textAnchor="middle" y={-4} fill="#e2e8f0" fontSize="2.2" fontFamily="monospace" fontWeight={600}>GDP: {n.gdp.toFixed(1)}B</text>
   252→   252→                </g>
   253→   253→              )}
   254→   254→            </g>
   255→   255→          );
   256→   256→        })}
   257→   257→      </svg>
   258→   258→      <div className="absolute bottom-3 left-4 text-[10px] text-slate-600 font-mono">Scroll: zoom · Drag: pan · Click: select</div>
   259→   259→    </div>
   260→   260→  );
   261→   261→}
