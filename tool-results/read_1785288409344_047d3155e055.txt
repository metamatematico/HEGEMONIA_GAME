    71→/** Social classes within a nation */

    79→/** Resource levels of a nation */

    83→  grain: number;     // 0-100
    84→  cotton: number;    // 0-100
    85→}

    87→/** A nation node in the world graph */
    88→export interface NationNode {
    89→  id: string;
    90→  name: string;
    91→  flag: string;
    92→  worldClass: WorldClass;
    93→  primaryIdeology: Ideology;
    94→  secondaryIdeologies: Ideology[];  // 0-2 additional ideologies (coalitions)
    95→
    96→  // Economy
    97→  gdp: number;           // billions (arbitrary scale)
    98→  population: number;    // millions
    99→  tradeBalance: number;  // surplus(+)/deficit(-) millions
   100→
   101→  // Power dimensions (0-100)
   102→  militaryPower: number;
   103→  culturalPower: number;
   104→  diplomaticPower: number;
   105→
   106→  // Graph position (0-100 for rendering)
   107→  x: number;
   108→  y: number;
   109→
   110→  // Graph metrics (computed each turn)
   111→  eigenvectorCentrality: number;  // 0-1
   112→  betweennessCentrality: number;  // 0-1
   113→
   114→  // Resources
   115→  resources: Resources;
   116→
   117→  // Social structure
   118→  socialClasses: SocialClasses;
   119→
   120→  // State for NPC AI
   121→  stability: number;      // 0-100, affects revolution risk
   122→  unrest: number;         // 0-100, rises when stability drops
   123→  industrialization: number; // 0-100, tech progress
   124→
   125→  // Player control
   126→  isPlayer: boolean;
   127→
   128→  // Cultural trait (NPC-inferred or player-chosen)
   129→  culturalTrait: CulturalTrait;
   130→}
   131→
   132→/** A trade edge in the world graph */
   133→export interface TradeEdge {
   134→  id: string;
   135→  from: string;     // nation id
   136→  to: string;       // nation id
   137→  volume: number;   // trade volume in millions
   138→  type: TradeType;
   139→  tariffRate: number; // 0-1, affected by ideology
   140→}

   195→/** Complete game state */
   196→export interface GameState {
   197→  turn: number;
   198→  year: number;
   199→  nations: NationNode[];

   210→

   212→  playerActionQueue: PlayerAction[];
   213→  playerCooldowns: CooldownMap;
   214→  playerActionsUsedThisTurn: number;
   215→  maxActionsPerTurn: number;
   216→

   227→  lowMilitaryTurns: number;
