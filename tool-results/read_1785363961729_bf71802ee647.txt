    97→// ─── GRAPH VIEW with full interactivity ───
    98→function GraphView() {
    99→  const svgRef = useRef<SVGSVGElement>(null);
   100→  const [hoveredNode, setHoveredNode] = useState<string | null>(null);
   101→  const [viewBox, setViewBox] = useState("-5 -5 110 90");
   102→  const [isDragging, setIsDragging] = useState(false);
   103→  const [layoutMode, setLayoutMode] = useState<GraphLayout>("original");
   104→  const [showLayoutMenu, setShowLayoutMenu] = useState(false);
   105→  const dragStart = useRef({ x: 0, y: 0 });
   106→  const vbStart = useRef({ x: -5, y: -5 });
   107→
   108→  const nations = useHegemoniaStore((s) => s.nations);
   109→  const edges = useHegemoniaStore((s) => s.edges);
   110→  const selectedNationId = useHegemoniaStore((s) => s.selectedNationId);
   111→  const setSelectedNation = useHegemoniaStore((s) => s.setSelectedNation);
   112→  const activeTab = useHegemoniaStore((s) => s.activeTab);
   113→  const actionTargetId = useHegemoniaStore((s) => s.actionTargetId);
   114→  const setActionTarget = useHegemoniaStore((s) => s.setActionTarget);
   115→
   116→  const maxVolume = useMemo(() => Math.max(1, ...edges.map((e) => e.volume)), [edges]);
   117→
   118→  // ─── Layout position overrides ───
   119→  const layoutPositions = useMemo(() => {
   120→    if (layoutMode === "original") return null;
   121→
   122→    const pos = new Map<string, { x: number; y: number }>();
   123→
   124→    if (layoutMode === "hierarchy") {
   125→      // Arrange by worldClass in rows: core on top, semi in middle, periphery on bottom
   126→      const byClass: Record<string, NationNode[]> = { core: [], semi: [], periphery: [] };
   127→      for (const n of nations) byClass[n.worldClass]?.push(n);
   128→      const rows: { nodes: NationNode[]; y: number }[] = [
   129→        { nodes: byClass.core || [], y: 15 },
   130→        { nodes: byClass.semi || [], y: 45 },
   131→        { nodes: byClass.periphery || [], y: 72 },
   132→      ];
   133→      for (const row of rows) {
   134→        const spacing = 90 / Math.max(1, row.nodes.length + 1);
   135→        row.nodes.forEach((n, i) => {
   136→          pos.set(n.id, { x: 5 + spacing * (i + 1), y: row.y });
   137→        });
   138→      }
   139→    } else if (layoutMode === "eigenvector" || layoutMode === "betweenness" || layoutMode === "pagerank") {
   140→      const scores = layoutMode === "eigenvector"
   141→        ? eigenvectorCentrality(nations, edges)
   142→        : layoutMode === "betweenness"
   143→          ? betweennessCentrality(nations, edges)
   144→          : pageRank(nations, edges);
   145→
   146→      // Sort by score descending, arrange in grid
   147→      const sorted = [...nations].sort((a, b) => (scores.get(b.id) ?? 0) - (scores.get(a.id) ?? 0));
   148→      const cols = Math.ceil(Math.sqrt(sorted.length));
   149→      const totalRows = Math.ceil(sorted.length / cols);
   150→      const colSpacing = cols > 1 ? 80 / (cols - 1) : 0;
   151→      const rowSpacing = totalRows > 1 ? 70 / (totalRows - 1) : 0;
   152→      sorted.forEach((n, i) => {
   153→        const col = i % cols;
   154→        const row = Math.floor(i / cols);
   155→        pos.set(n.id, {
   156→          x: 5 + colSpacing * col,
   157→          y: 5 + rowSpacing * row,
   158→        });
   159→      });
   160→    } else if (layoutMode === "gdp") {
   161→      const sorted = [...nations].sort((a, b) => b.gdp - a.gdp);
   162→      const cols = Math.ceil(Math.sqrt(sorted.length));
   163→      const totalRows = Math.ceil(sorted.length / cols);
   164→      const colSpacing = cols > 1 ? 80 / (cols - 1) : 0;
   165→      const rowSpacing = totalRows > 1 ? 70 / (totalRows - 1) : 0;
   166→      sorted.forEach((n, i) => {
   167→        const col = i % cols;
   168→        const row = Math.floor(i / cols);
   169→        pos.set(n.id, {
   170→          x: 5 + colSpacing * col,
   171→          y: 5 + rowSpacing * row,
   172→        });
   173→      });
   174→    }
   175→
   176→    return pos;
   177→  }, [layoutMode, nations, edges]);
   178→
   179→  // Helper: get position for a nation (layout override or original)
   180→  const getNodePos = useCallback((n: NationNode) => {
   181→    if (layoutPositions) {
   182→      const p = layoutPositions.get(n.id);
   183→      if (p) return p;
   184→    }
   185→    return { x: n.x, y: n.y };
   186→  }, [layoutPositions]);
   187→
   188→  // Zoom centered on cursor
   189→  const handleWheel = useCallback((e: React.WheelEvent) => {
   190→    e.preventDefault();
   191→    const [vx, vy, vw, vh] = viewBox.split(" ").map(Number);
   192→    const svg = svgRef.current;
   193→    if (!svg) return;
   194→    const rect = svg.getBoundingClientRect();
   195→    const mx = ((e.clientX - rect.left) / rect.width) * vw + vx;
   196→    const my = ((e.clientY - rect.top) / rect.height) * vh + vy;
   197→    const f = e.deltaY > 0 ? 1.1 : 0.9;
   198→    const nvw = Math.max(20, Math.min(150, vw * f));
   199→    const nvh = nvw * (vh / vw);
   200→    const r = nvw / vw;
   201→    setViewBox(`${(mx - (mx - vx) * r).toFixed(1)} ${(my - (my - vy) * r).toFixed(1)} ${nvw.toFixed(1)} ${nvh.toFixed(1)}`);
   202→  }, [viewBox]);
   203→
   204→  // Pan
   205→  const handleMouseDown = useCallback((e: React.MouseEvent) => {
   206→    if ((e.target as Element).closest(".gn")) return;
   207→    setIsDragging(true);
   208→    dragStart.current = { x: e.clientX, y: e.clientY };
   209→    const [vx, vy] = viewBox.split(" ").map(Number);
   210→    vbStart.current = { x: vx, y: vy };
   211→  }, [viewBox]);
   212→
   213→  const handleMouseMove = useCallback((e: React.MouseEvent) => {
   214→    if (!isDragging) return;
   215→    const svg = svgRef.current;
   216→    if (!svg) return;
   217→    const rect = svg.getBoundingClientRect();
   218→    const [vw, , , vh] = viewBox.split(" ").map(Number);
   219→    const sx = vw / rect.width;
   220→    const sy = sx;
   221→    setViewBox(`${(vbStart.current.x - (e.clientX - dragStart.current.x) * sx).toFixed(1)} ${(vbStart.current.y - (e.clientY - dragStart.current.y) * sy).toFixed(1)} ${vw} ${vh}`);
   222→  }, [isDragging, viewBox]);
   223→
   224→  // Reset view
   225→  const resetView = useCallback(() => setViewBox("-5 -5 110 90"), []);
   226→
   227→  const highlightNode = selectedNationId || hoveredNode;
   228→
   229→  return (
   230→    <div className="relative w-full h-full min-h-[300px] bg-[#070b14] rounded-lg overflow-hidden border border-slate-800/50">
   231→      {/* Grid background */}
   232→      <svg className="absolute inset-0 w-full h-full opacity-[0.04] pointer-events-none">
   233→        <defs><pattern id="grid" width="5" height="5" patternUnits="userSpaceOnUse"><path d="M 5 0 L 0 0 0 5" fill="none" stroke="#06b6d4" strokeWidth="0.15" /></pattern></defs>
   234→        <rect width="100%" height="100%" fill="url(#grid)" />
   235→      </svg>
   236→
   237→      {/* Top bar */}
   238→      <div className="absolute top-3 left-4 z-10 flex items-center gap-2" onClick={(e) => e.stopPropagation()}>
   239→        <div className="w-2 h-2 rounded-full bg-cyan-400 animate-pulse" />
   240→        <span className="text-xs font-mono text-cyan-400/70 uppercase tracking-wider">Grafo Mundial — {nations.length} nodos, {edges.length} aristas</span>
   241→      </div>
   242→
   243→      {/* Legend */}
   244→      <div className="absolute top-3 right-3 z-10 flex flex-col gap-1.5 bg-black/40 backdrop-blur-sm rounded px-3 py-2 border border-slate-800/30" onClick={(e) => e.stopPropagation()}>
   245→        {(["core", "semi", "periphery"] as const).map((c) => (
   246→          <div key={c} className="flex items-center gap-2">
   247→            <div className="w-2.5 h-2.5 rounded-full" style={{ background: classColor(c) }} />
   248→            <span className="text-[10px] text-slate-400">{classLabel(c)}</span>
   249→          </div>
   250→        ))}
   251→      </div>
   252→
   253→      {/* Controls bar */}
   254→      <div className="absolute bottom-3 right-3 z-10 flex gap-1.5 items-center" onClick={(e) => e.stopPropagation()}>
   255→        {/* Layout selector */}
   256→        <div className="relative">
   257→          <button
   258→            onClick={(e) => { e.stopPropagation(); e.preventDefault(); setShowLayoutMenu(!showLayoutMenu); }}
   259→            className={`h-7 px-2 rounded bg-black/40 backdrop-blur-sm border border-slate-800/30 flex items-center gap-1 hover:bg-slate-800/50 transition-colors text-[10px] font-mono ${layoutMode !== "original" ? "text-cyan-400 border-cyan-400/40" : "text-slate-400"}`}
   260→            title="Distribución del grafo"
   261→          >
   262→            <Network className="w-3 h-3" />
   263→            <span className="hidden sm:inline">{LAYOUT_DEFS.find(l => l.id === layoutMode)?.label}</span>
   264→          </button>
   265→          {showLayoutMenu && (
   266→            <div className="absolute bottom-9 right-0 bg-[#111827] border border-slate-700/50 rounded-lg shadow-xl p-1 min-w-[180px] z-50" onClick={(e) => e.stopPropagation()}>
   267→              {LAYOUT_DEFS.map((l) => (
   268→                <button
   269→                  key={l.id}
   270→                  onClick={(e) => { e.stopPropagation(); e.preventDefault(); setLayoutMode(l.id); setShowLayoutMenu(false); }}
   271→                  className={`w-full px-3 py-2 rounded text-left text-[11px] font-mono flex items-center gap-2 transition-colors ${
   272→                    layoutMode === l.id ? "bg-cyan-400/15 text-cyan-400" : "text-slate-400 hover:bg-slate-800/50 hover:text-slate-200"
   273→                  }`}
   274→                >
   275→                  <l.icon className="w-3.5 h-3.5 shrink-0" />
   276→                  <div>
   277→                    <div className="font-semibold">{l.label}</div>
   278→                    <div className="text-[9px] text-slate-500">{l.desc}</div>
   279→                  </div>
   280→                </button>
   281→              ))}
   282→            </div>
   283→          )}
   284→        </div>
   285→        <button onClick={(e) => { e.stopPropagation(); resetView(); }}
   286→          className="w-7 h-7 rounded bg-black/40 backdrop-blur-sm border border-slate-800/30 flex items-center justify-center hover:bg-slate-800/50 transition-colors"
   287→          title="Resetear vista">
   288→          <Maximize2 className="w-3 h-3 text-slate-400" />
   289→        </button>
   290→        <button onClick={(e) => { e.stopPropagation(); setViewBox(vb => { const [x,y,w,h] = vb.split(" ").map(Number); return `${x.toFixed(1)} ${y.toFixed(1)} ${(w*0.7).toFixed(1)} ${(h*0.7).toFixed(1)}`; }); }}
   291→          className="w-7 h-7 rounded bg-black/40 backdrop-blur-sm border border-slate-800/30 flex items-center justify-center hover:bg-slate-800/50 transition-colors text-[11px] text-slate-400 font-bold"
   292→          title="Zoom +">
   293→          +
   294→        </button>
   295→        <button onClick={(e) => { e.stopPropagation(); setViewBox(vb => { const [x,y,w,h] = vb.split(" ").map(Number); return `${(x-w*0.15).toFixed(1)} ${(y-h*0.15).toFixed(1)} ${(w*1.3).toFixed(1)} ${(h*1.3).toFixed(1)}`; }); }}
   296→          className="w-7 h-7 rounded bg-black/40 backdrop-blur-sm border border-slate-800/30 flex items-center justify-center hover:bg-slate-800/50 transition-colors text-[11px] text-slate-400 font-bold"
   297→          title="Zoom -">
   298→          −
   299→        </button>
   300→      </div>
   301→
   302→      {/* SVG Graph */}
   303→      <svg ref={svgRef} viewBox={viewBox} className="w-full h-full cursor-grab active:cursor-grabbing"
   304→        onWheel={handleWheel} onMouseDown={handleMouseDown} onMouseMove={handleMouseMove}
   305→        onMouseUp={() => setIsDragging(false)} onMouseLeave={() => setIsDragging(false)}
   306→        onClick={(e) => { if (!(e.target as Element).closest(".gn")) setSelectedNation(null); }}
   307→        style={{ touchAction: "none" }}>
   308→        <defs>
   309→          <filter id="gc" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="2" result="b" /><feFlood floodColor="#06b6d4" floodOpacity="0.6" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   310→          <filter id="gs" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="1.5" result="b" /><feFlood floodColor="#f59e0b" floodOpacity="0.5" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   311→          <filter id="gk" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="3" result="b" /><feFlood floodColor="#06b6d4" floodOpacity="0.8" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   312→        </defs>
   313→
   314→        {/* Edges */}
   315→        {edges.map((e, i) => {
   316→          const from = nations.find((n) => n.id === e.from);
   317→          const to = nations.find((n) => n.id === e.to);
   318→          if (!from || !to) return null;
   319→          const fp = getNodePos(from);
   320→          const tp = getNodePos(to);
   321→          const isHi = highlightNode ? (e.from === highlightNode || e.to === highlightNode) : false;
   322→          const op = highlightNode ? (isHi ? 0.7 : 0.05) : 0.15 + (e.volume / maxVolume) * 0.25;
   323→          const sc = isHi ? (e.type === "raw" ? "#10b981" : e.type === "manufactured" ? "#06b6d4" : "#a855f7") : "#334155";
   324→          return (
   325→            <line key={i} x1={fp.x} y1={fp.y} x2={tp.x} y2={tp.y}
   326→              stroke={sc} strokeWidth={isHi ? Math.min(3, e.volume / 100) * 1.5 : Math.min(3, e.volume / 100)}
   327→              opacity={op} strokeLinecap="round">
   328→              {isHi && <animate attributeName="stroke-dasharray" values="2 6;6 2;2 6" dur="2s" repeatCount="indefinite" />}
   329→            </line>
   330→          );
   331→        })}
   332→
   333→        {/* Nodes */}
   334→        {nations.map((n) => {
   335→          const { x: nx, y: ny } = getNodePos(n);
   336→          const r = Math.max(3, 3 + (n.gdp / 4.5) * 7);
   337→          const isSel = n.id === selectedNationId;
   338→          const isHov = n.id === hoveredNode;
   339→          const isConn = highlightNode && edges.some((e) => (e.from === highlightNode && e.to === n.id) || (e.to === highlightNode && e.from === n.id));
   340→          const dimmed = highlightNode && !isSel && !isConn && n.id !== highlightNode;
   341→          const color = classColor(n.worldClass);
   342→          return (
   343→            <g key={n.id} className="gn" transform={`translate(${nx},${ny})`} opacity={dimmed ? 0.15 : 1}
   344→              style={{ cursor: "pointer" }}
   345→              onClick={(ev) => {
   346→                ev.stopPropagation();
   347→                if (activeTab === "actions" && n.id !== "gb") {
   348→                  setActionTarget(isSel ? null : n.id);
   349→                } else {
   350→                  setSelectedNation(isSel ? null : n.id);
   351→                }
   352→              }}
   353→              onMouseEnter={() => setHoveredNode(n.id)} onMouseLeave={() => setHoveredNode(null)}>
   354→              {isSel && <circle r={r + 4} fill="none" stroke={color} strokeWidth="1.5" opacity="0.4"><animate attributeName="r" values={`${r + 3};${r + 6};${r + 3}`} dur="2s" repeatCount="indefinite" /><animate attributeName="opacity" values="0.4;0.15;0.4" dur="2s" repeatCount="indefinite" /></circle>}
   355→              {n.id === actionTargetId && (
   356→                <circle r={r + 3} fill="none" stroke="#ef4444" strokeWidth="1.2" strokeDasharray="3 2" opacity="0.8">
   357→                  <animateTransform attributeName="transform" type="rotate" from="0" to="360" dur="6s" repeatCount="indefinite" />
   358→                </circle>
   359→              )}
   360→              <circle r={r} fill={color} opacity={isSel || isHov ? 1 : 0.75}

   607→// ─── Player Actions Panel ───
   608→function PlayerActionsPanel() {
   609→  const gameState = useHegemoniaStore((s) => s.gameState);
   610→  const nations = useHegemoniaStore((s) => s.nations);
   611→  const queuePlayerAction = useHegemoniaStore((s) => s.queuePlayerAction);
   612→  const actionTargetId = useHegemoniaStore((s) => s.actionTargetId);
   613→  const setActionTarget = useHegemoniaStore((s) => s.setActionTarget);
   614→  const lastResult = useHegemoniaStore((s) => s.lastActionResult);
   615→  const clearActionResult = useHegemoniaStore((s) => s.clearActionResult);
   616→
   617→  const player = nations.find((n) => n.isPlayer);
   618→  if (!player) return <div className="p-4 text-slate-600 text-xs">No hay nación del jugador</div>;
   619→
   620→  const actionsLeft = gameState.maxActionsPerTurn - gameState.playerActionsUsedThisTurn;
   621→  const cooldowns = gameState.playerCooldowns;
   622→  const target = actionTargetId ? nations.find((n) => n.id === actionTargetId) : undefined;
   623→
   624→  // Group actions by category
   625→  const categories: Record<string, typeof PLAYER_ACTIONS> = { economy: [], military: [], diplomacy: [], ideology: [], control: [] };
   626→  for (const a of PLAYER_ACTIONS) categories[a.category].push(a);
   627→  const categoryLabels: Record<string, string> = { economy: "Economía", military: "Militar", diplomacy: "Diplomacia", ideology: "Ideología", control: "Control" };
   628→
   629→  const handleAction = (type: PlayerActionType, requiresTarget: boolean) => {
   630→    if (requiresTarget && !actionTargetId) return;
   631→    queuePlayerAction(type, requiresTarget ? actionTargetId ?? undefined : undefined);
   632→    setTimeout(() => clearActionResult(), 3000);
   633→  };
   634→
   635→  return (
   636→    <div className="h-full flex flex-col overflow-hidden">
   637→      <div className="px-3 py-2 border-b border-slate-800/50 bg-slate-900/30 flex items-center gap-3 shrink-0">

   648→              <div key={i} className={`w-3 h-3 rounded-sm ${i < actionsLeft ? "bg-cyan-400" : "bg-slate-700"}`} />
   649→            ))}

   651→          <span className="text-[9px] text-slate-500 font-mono mt-0.5">

   660→        }`}>

   665→      <div className="px-3 py-1.5 border-b border-slate-800/30 shrink-0 flex items-center gap-2">
