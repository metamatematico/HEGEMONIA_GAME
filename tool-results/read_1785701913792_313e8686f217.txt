   665→// ─── Player Actions Panel ───
   666→function PlayerActionsPanel() {
   667→  const gameState = useHegemoniaStore((s) => s.gameState);
   668→  const nations = useHegemoniaStore((s) => s.nations);
   669→  const queuePlayerAction = useHegemoniaStore((s) => s.queuePlayerAction);
   670→  const actionTargetId = useHegemoniaStore((s) => s.actionTargetId);
   671→  const setActionTarget = useHegemoniaStore((s) => s.setActionTarget);
   672→  const lastResult = useHegemoniaStore((s) => s.lastActionResult);
   673→  const clearActionResult = useHegemoniaStore((s) => s.clearActionResult);
   674→
   675→  const player = nations.find((n) => n.isPlayer);
   676→  if (!player) return <div className="p-4 text-slate-600 text-xs">No hay nación del jugador</div>;
   677→
   678→  const actionsLeft = gameState.maxActionsPerTurn - gameState.playerActionsUsedThisTurn;
   679→  const cooldowns = gameState.playerCooldowns;
   680→  const target = actionTargetId ? nations.find((n) => n.id === actionTargetId) : undefined;
   681→
   682→  // Group actions by category
   683→  const categories: Record<string, typeof PLAYER_ACTIONS> = { economy: [], military: [], diplomacy: [], ideology: [], control: [] };
   684→  for (const a of PLAYER_ACTIONS) categories[a.category].push(a);
   685→  const categoryLabels: Record<string, string> = { economy: "Economía", military: "Militar", diplomacy: "Diplomacia", ideology: "Ideología", control: "Control" };
   686→
   687→  const handleAction = (type: PlayerActionType, requiresTarget: boolean) => {
   688→    if (requiresTarget && !actionTargetId) return;
   689→    queuePlayerAction(type, requiresTarget ? actionTargetId ?? undefined : undefined);
   690→    setTimeout(() => clearActionResult(), 3000);
   691→  };
   692→
   693→  return (
   694→    <div className="h-full flex flex-col overflow-hidden">
   695→      <div className="px-3 py-2 border-b border-slate-800/50 bg-slate-900/30 flex items-center gap-3 shrink-0">
   696→        <span className="text-base">{player.flag}</span>
   697→        <div className="flex-1 min-w-0">
   698→          <div className="text-xs text-white font-semibold">{player.name}</div>
   699→          <div className="text-[10px] text-slate-500 font-mono">
   700→            PIB: ${player.gdp.toFixed(2)}B · Estab: {player.stability.toFixed(0)}% · Descont: {player.unrest.toFixed(0)}%
   701→          </div>
   702→        </div>
   703→        <div className="flex flex-col items-end">
   704→          <div className="flex gap-0.5">
   705→            {Array.from({ length: gameState.maxActionsPerTurn }).map((_, i) => (
   706→              <div key={i} className={`w-3 h-3 rounded-sm ${i < actionsLeft ? "bg-cyan-400" : "bg-slate-700"}`} />
   707→            ))}
   708→          </div>
   709→          <span className="text-[9px] text-slate-500 font-mono mt-0.5">
   710→            {actionsLeft} acción{actionsLeft !== 1 ? "es" : ""} restante{actionsLeft !== 1 ? "s" : ""}
   711→          </span>
   712→        </div>
   713→      </div>
   714→
   715→      {lastResult && (
   716→        <div className={`px-3 py-1.5 text-[10px] font-mono border-b shrink-0 ${
   717→          lastResult.success ? "bg-emerald-500/10 text-emerald-400 border-emerald-500/20" : "bg-red-500/10 text-red-400 border-red-500/20"
   718→        }`}>
   719→          {lastResult.success ? "✓ Acción encolada" : `✗ ${lastResult.reason ?? "Error"}`}
   720→        </div>
   721→      )}
   722→
   723→      <div className="px-3 py-1.5 border-b border-slate-800/30 shrink-0 flex items-center gap-2">
   724→        <span className="text-[9px] uppercase tracking-wider text-slate-600 font-mono">Objetivo:</span>
   725→        {target ? (
   726→          <>
   727→            <span className="text-xs">{target.flag}</span>
   728→            <span className="text-[11px] text-slate-300 flex-1 truncate">{target.name}</span>
   729→            <button onClick={() => setActionTarget(null)} className="text-[10px] text-slate-500 hover:text-red-400 font-mono">✕</button>
   730→          </>
   731→        ) : (
   732→          <span className="text-[10px] text-slate-600 italic flex-1">Selecciona una nación en el grafo para acciones con objetivo</span>
   733→        )}
   734→      </div>
   735→
   736→      <div className="flex-1 overflow-y-auto heg-scroll p-2 space-y-2">
   737→        {Object.entries(categories).map(([catKey, actions]) => (
   738→          actions.length > 0 && (
   739→            <div key={catKey}>
   740→              <div className="text-[9px] uppercase tracking-wider text-slate-600 font-mono mb-1 px-1">{categoryLabels[catKey]}</div>
   741→              <div className="grid grid-cols-2 gap-1.5">
   742→                {actions.map((action) => {
   743→                  const meta = getPlayerActionMeta(action.type);
   744→                  const onCd = isOnCooldown(action.type, cooldowns, gameState.turn);
   745→                  const affordable = canAfford(action.type, player);
   746→                  const cost = getEffectiveCost(action.type, player);
   747→                  const needsTarget = meta.requiresTarget && !actionTargetId;
   748→                  const disabled = onCd || !affordable || needsTarget || actionsLeft <= 0;
   749→                  const hasIdeologyBonus = meta.ideologyBonus?.includes(player.primaryIdeology);
   750→                  return (
   751→                    <button key={action.type} onClick={() => handleAction(action.type, meta.requiresTarget)} disabled={disabled}
   752→                      className={`text-left p-2 rounded border transition-all relative ${
   753→                        disabled ? "bg-slate-900/30 border-slate-800/30 opacity-50 cursor-not-allowed" : "bg-slate-800/40 border-slate-700/50 hover:bg-cyan-400/10 hover:border-cyan-400/30 cursor-pointer"
   754→                      }`}>
   755→                      <div className="flex items-center gap-1.5 mb-0.5">
   756→                        <span className="text-sm">{action.icon}</span>
   757→                        <span className="text-[11px] font-semibold text-slate-200 truncate">{action.label}</span>
   758→                        {hasIdeologyBonus && <span className="text-[8px] px-1 rounded bg-cyan-400/20 text-cyan-400 font-mono ml-auto shrink-0">BONUS</span>}

   761→                      <div className="flex items-center justify-between text-[9px] font-mono">
   762→                        <span className={affordable ? "text-amber-400" : "text-red-400"}>${cost}B</span>
   763→                        {onCd && <span className="text-slate-500">⏱ {(cooldowns[action.type] ?? 0) - gameState.turn}T</span>}

   766→                  );

   768→              </div>
   769→            </div>
   770→          )
   771→        ))}
