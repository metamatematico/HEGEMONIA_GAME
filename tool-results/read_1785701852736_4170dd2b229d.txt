    97→// ─── Error Boundary ───
    98→class GraphErrorBoundary extends React.Component<React.PropsWithChildren<{ fallback?: React.ReactNode }>, { hasError: boolean; error?: Error }> {
    99→  constructor(props: React.PropsWithChildren<{ fallback?: React.ReactNode }>) {
   100→    super(props);
   101→    this.state = { hasError: false, error: undefined };
   102→  }
   103→  static getDerivedStateFromError(error: Error) {
   104→    return { hasError: true, error };
   105→  }
   106→  render() {
   107→    if (this.state.hasError) {
   108→      return this.props.fallback ?? (
   109→        <div className="flex items-center justify-center h-full bg-red-950/30 text-red-400 text-xs p-4 text-center">
   110→          <div>
   111→            <p className="font-bold mb-1">Error en el grafo</p>
   112→            <p className="text-red-300 text-[10px] font-mono">{this.state.error?.message}</p>
   113→          </div>
   114→        </div>
   115→      );
   116→    }
   117→    return this.props.children;
   118→  }
   119→}

   121→// ─── GRAPH VIEW with full interactivity ───
   122→function GraphView() {
   123→  const svgRef = useRef<SVGSVGElement>(null);
   124→  const [hoveredNode, setHoveredNode] = useState<string | null>(null);
   125→  const [viewBox, setViewBox] = useState("-5 -5 110 90");
   126→  const [isDragging, setIsDragging] = useState(false);
   127→  const [layoutMode, setLayoutMode] = useState<GraphLayout>("original");
   128→  const [showLayoutMenu, setShowLayoutMenu] = useState(false);
   129→  const [layoutError, setLayoutError] = useState<string | null>(null);
   130→  const [, startLayoutTransition] = useTransition();
   131→  const dragStart = useRef({ x: 0, y: 0 });
   132→  const vbStart = useRef({ x: -5, y: -5 });
   133→
   134→  const nations = useHegemoniaStore((s) => s.nations);
   135→  const edges = useHegemoniaStore((s) => s.edges);
   136→  const selectedNationId = useHegemoniaStore((s) => s.selectedNationId);
   137→  const setSelectedNation = useHegemoniaStore((s) => s.setSelectedNation);
   138→  const activeTab = useHegemoniaStore((s) => s.activeTab);
   139→  const actionTargetId = useHegemoniaStore((s) => s.actionTargetId);
   140→  const setActionTarget = useHegemoniaStore((s) => s.setActionTarget);
   141→
   142→  const maxVolume = useMemo(() => Math.max(1, ...edges.map((e) => e.volume)), [edges]);
   143→
   144→  // ─── Layout position overrides ───
   145→  const layoutPositions = useMemo(() => {
   146→    if (layoutMode === "original") return null;
   147→
   148→    const pos = new Map<string, { x: number; y: number }>();
   149→
   150→    try {
   151→      if (layoutMode === "hierarchy") {
   152→        // Arrange by worldClass in rows: core on top, semi in middle, periphery on bottom
   153→        const byClass: Record<string, NationNode[]> = { core: [], semi: [], periphery: [] };
   154→        for (const n of nations) {
   155→          const cls = n.worldClass || "periphery";
   156→          if (!byClass[cls]) byClass[cls] = [];
   157→          byClass[cls].push(n);
   158→        }
   159→        const rows: { nodes: NationNode[]; y: number }[] = [
   160→          { nodes: byClass.core || [], y: 15 },
   161→          { nodes: byClass.semi || [], y: 45 },
   162→          { nodes: byClass.periphery || [], y: 72 },
   163→        ];
   164→        for (const row of rows) {
   165→          const spacing = 90 / Math.max(1, row.nodes.length + 1);
   166→          row.nodes.forEach((n, i) => {
   167→            pos.set(n.id, { x: 5 + spacing * (i + 1), y: row.y });
   168→          });
   169→        }
   170→      } else if (layoutMode === "eigenvector" || layoutMode === "betweenness" || layoutMode === "pagerank") {
   171→        const scores = layoutMode === "eigenvector"
   172→          ? eigenvectorCentrality(nations, edges)
   173→          : layoutMode === "betweenness"
   174→            ? betweennessCentrality(nations, edges)
   175→            : pageRank(nations, edges);
   176→
   177→        // Sort by score descending, arrange in grid
   178→        const sorted = [...nations].sort((a, b) => (scores.get(b.id) ?? 0) - (scores.get(a.id) ?? 0));
   179→        const cols = Math.max(1, Math.ceil(Math.sqrt(sorted.length)));
   180→        const totalRows = Math.ceil(sorted.length / cols);
   181→        const colSpacing = cols > 1 ? 80 / (cols - 1) : 0;
   182→        const rowSpacing = totalRows > 1 ? 70 / (totalRows - 1) : 0;
   183→        sorted.forEach((n, i) => {
   184→          const col = i % cols;
   185→          const row = Math.floor(i / cols);
   186→          pos.set(n.id, {
   187→            x: 5 + colSpacing * col,
   188→            y: 5 + rowSpacing * row,
   189→          });
   190→        });
   191→      } else if (layoutMode === "gdp") {
   192→        const sorted = [...nations].sort((a, b) => b.gdp - a.gdp);
   193→        const cols = Math.max(1, Math.ceil(Math.sqrt(sorted.length)));
   194→        const totalRows = Math.ceil(sorted.length / cols);
   195→        const colSpacing = cols > 1 ? 80 / (cols - 1) : 0;
   196→        const rowSpacing = totalRows > 1 ? 70 / (totalRows - 1) : 0;
   197→        sorted.forEach((n, i) => {
   198→          const col = i % cols;
   199→          const row = Math.floor(i / cols);
   200→          pos.set(n.id, {
   201→            x: 5 + colSpacing * col,
   202→            y: 5 + rowSpacing * row,
   203→          });
   204→        });
   205→      }
   206→    } catch (err) {
   207→      console.error("[Hegemonia] Layout calculation error:", err);
   208→      return null;
   209→    }
   210→
   211→    return pos;
   212→  }, [layoutMode, nations, edges]);
   213→
   214→  // Detect layout errors for UI feedback
   215→  useEffect(() => {
   216→    if (layoutMode !== "original" && !layoutPositions) {
   217→      setLayoutError(`No se pudo calcular el layout "${layoutMode}"`);
   218→    } else {
   219→      setLayoutError(null);
   220→    }
   221→  }, [layoutMode, layoutPositions]);
   222→
   223→  // Helper: get position for a nation (layout override or original)
   224→  const getNodePos = useCallback((n: NationNode) => {
   225→    if (layoutPositions) {
   226→      const p = layoutPositions.get(n.id);
   227→      if (p && isFinite(p.x) && isFinite(p.y)) return p;
   228→    }
   229→    const ox = typeof n.x === "number" && isFinite(n.x) ? n.x : 50;
   230→    const oy = typeof n.y === "number" && isFinite(n.y) ? n.y : 50;
   231→    return { x: ox, y: oy };
   232→  }, [layoutPositions]);
   233→
   234→  // Zoom centered on cursor — uses native event listener (non-passive) to avoid passive event error
   235→  useEffect(() => {
   236→    const svg = svgRef.current;
   237→    if (!svg) return;
   238→    const handler = (e: WheelEvent) => {
   239→      e.preventDefault();
   240→      const vb = viewBox.split(" ").map(Number);
   241→      const [vx, vy, vw, vh] = vb;
   242→      const rect = svg.getBoundingClientRect();
   243→      const mx = ((e.clientX - rect.left) / rect.width) * vw + vx;
   244→      const my = ((e.clientY - rect.top) / rect.height) * vh + vy;
   245→      const f = e.deltaY > 0 ? 1.1 : 0.9;
   246→      const nvw = Math.max(20, Math.min(150, vw * f));
   247→      const nvh = nvw * (vh / vw);
   248→      const r = nvw / vw;
   249→      setViewBox(`${(mx - (mx - vx) * r).toFixed(1)} ${(my - (my - vy) * r).toFixed(1)} ${nvw.toFixed(1)} ${nvh.toFixed(1)}`);
   250→    };
   251→    svg.addEventListener("wheel", handler, { passive: false });
   252→    return () => { svg.removeEventListener("wheel", handler); };
   253→  }, [viewBox]);
   254→
   255→  // Pan
   256→  const handleMouseDown = useCallback((e: React.MouseEvent) => {
   257→    if ((e.target as Element).closest(".gn")) return;
   258→    setIsDragging(true);
   259→    dragStart.current = { x: e.clientX, y: e.clientY };
   260→    const [vx, vy] = viewBox.split(" ").map(Number);
   261→    vbStart.current = { x: vx, y: vy };
   262→  }, [viewBox]);
   263→
   264→  const handleMouseMove = useCallback((e: React.MouseEvent) => {
   265→    if (!isDragging) return;
   266→    const svg = svgRef.current;
   267→    if (!svg) return;
   268→    const rect = svg.getBoundingClientRect();
   269→    const [vw, , , vh] = viewBox.split(" ").map(Number);
   270→    const sx = vw / rect.width;
   271→    const sy = sx;
   272→    setViewBox(`${(vbStart.current.x - (e.clientX - dragStart.current.x) * sx).toFixed(1)} ${(vbStart.current.y - (e.clientY - dragStart.current.y) * sy).toFixed(1)} ${vw} ${vh}`);
   273→  }, [isDragging, viewBox]);
   274→
   275→  // Reset view
   276→  const resetView = useCallback(() => setViewBox("-5 -5 110 90"), []);
   277→
   278→  const highlightNode = selectedNationId || hoveredNode;
   279→
   280→  return (
   281→    <div className="relative w-full h-full min-h-[300px] bg-[#070b14] rounded-lg overflow-hidden border border-slate-800/50">
   282→      {/* Grid background */}
   283→      <svg className="absolute inset-0 w-full h-full opacity-[0.04] pointer-events-none">
   284→        <defs><pattern id="grid" width="5" height="5" patternUnits="userSpaceOnUse"><path d="M 5 0 L 0 0 0 5" fill="none" stroke="#06b6d4" strokeWidth="0.15" /></pattern></defs>
   285→        <rect width="100%" height="100%" fill="url(#grid)" />
   286→      </svg>
   287→
   288→      {/* Top bar */}
   289→      <div className="absolute top-3 left-4 z-10 flex items-center gap-2" onClick={(e) => e.stopPropagation()}>
   290→        <div className="w-2 h-2 rounded-full bg-cyan-400 animate-pulse" />
   291→        <span className="text-xs font-mono text-cyan-400/70 uppercase tracking-wider">Grafo Mundial — {nations.length} nodos, {edges.length} aristas</span>
   292→      </div>
   293→
   294→      {/* Legend */}
   295→      <div className="absolute top-3 right-3 z-10 flex flex-col gap-1.5 bg-black/40 backdrop-blur-sm rounded px-3 py-2 border border-slate-800/30" onClick={(e) => e.stopPropagation()}>
   296→        {(["core", "semi", "periphery"] as const).map((c) => (
   297→          <div key={c} className="flex items-center gap-2">
   298→            <div className="w-2.5 h-2.5 rounded-full" style={{ background: classColor(c) }} />
   299→            <span className="text-[10px] text-slate-400">{classLabel(c)}</span>
   300→          </div>
   301→        ))}
   302→      </div>
   303→
   304→      {/* Controls bar */}
   305→      <div className="absolute bottom-3 right-3 z-10 flex gap-1.5 items-center" onClick={(e) => e.stopPropagation()}>
   306→        {/* Layout selector */}
   307→        <div className="relative">
   308→          <button
   309→            onClick={(e) => { e.stopPropagation(); e.preventDefault(); setShowLayoutMenu(!showLayoutMenu); }}
   310→            className={`h-7 px-2 rounded bg-black/40 backdrop-blur-sm border border-slate-800/30 flex items-center gap-1 hover:bg-slate-800/50 transition-colors text-[10px] font-mono ${layoutMode !== "original" ? "text-cyan-400 border-cyan-400/40" : "text-slate-400"}`}
   311→            title="Distribución del grafo"
   312→          >
   313→            <Network className="w-3 h-3" />
   314→            <span className="hidden sm:inline">{LAYOUT_DEFS.find(l => l.id === layoutMode)?.label}</span>
   315→          </button>
   316→          {showLayoutMenu && (

   323→                    layoutMode === l.id ? "bg-cyan-400/15 text-cyan-400" : "text-slate-400 hover:bg-slate-800/50 hover:text-slate-200"
   324→                  }`}

   331→                </button>
   332→              ))}
   333→            </div>
   334→          )}
   335→        </div>
   336→        <button onClick={(e) => { e.stopPropagation(); resetView(); }}
   337→          className="w-7 h-7 rounded bg-black/40 backdrop-blur-sm border border-slate-800/30 flex items-center justify-center hover:bg-slate-800/50 transition-colors"
   338→          title="Resetear vista">

   341→        <button onClick={(e) => { e.stopPropagation(); setViewBox(vb => { const [x,y,w,h] = vb.split(" ").map(Number); return `${x.toFixed(1)} ${y.toFixed(1)} ${(w*0.7).toFixed(1)} ${(h*0.7).toFixed(1)}`; }); }}
   342→          className="w-7 h-7 rounded bg-black/40 backdrop-blur-sm border border-slate-800/30 flex items-center justify-center hover:bg-slate-800/50 transition-colors text-[11px] text-slate-400 font-bold"
   343→          title="Zoom +">

   347→          className="w-7 h-7 rounded bg-black/40 backdrop-blur-sm border border-slate-800/30 flex items-center justify-center hover:bg-slate-800/50 transition-colors text-[11px] text-slate-400 font-bold"
   348→          title="Zoom -">
   349→          −
   350→        </button>
   351→      </div>
   352→
   353→      {/* Layout error banner */}
   354→      {layoutError && (
   355→        <div className="absolute top-10 left-1/2 -translate-x-1/2 z-20 bg-red-950/80 border border-red-500/30 rounded px-3 py-1.5 text-[10px] text-red-400 font-mono">
   356→          Error en layout: {layoutError} — <button onClick={() => { setLayoutMode("original"); setLayoutError(null); }} className="underline ml-1">Volver a Original</button>
   357→        </div>
   358→      )}
   359→
   360→      {/* SVG Graph */}
   361→      <svg ref={svgRef} viewBox={viewBox} className="w-full h-full cursor-grab active:cursor-grabbing"
   362→        onMouseDown={handleMouseDown} onMouseMove={handleMouseMove}
   363→        onMouseUp={() => setIsDragging(false)} onMouseLeave={() => setIsDragging(false)}
   364→        onClick={(e) => { if (!(e.target as Element).closest(".gn")) setSelectedNation(null); }}
   365→        style={{ touchAction: "none" }}>
   366→        <defs>
   367→          <filter id="gc" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="2" result="b" /><feFlood floodColor="#06b6d4" floodOpacity="0.6" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   368→          <filter id="gs" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="1.5" result="b" /><feFlood floodColor="#f59e0b" floodOpacity="0.5" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   369→          <filter id="gk" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="3" result="b" /><feFlood floodColor="#06b6d4" floodOpacity="0.8" result="c" /><feComposite in="c" in2="b" operator="in" result="g" /><feMerge><feMergeNode in="g" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
   370→        </defs>
   371→
   372→        {/* Edges */}
